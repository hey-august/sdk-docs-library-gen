1//! Webhook signature validation for SignalWire-signed HTTP requests.
2//!
3//! Implements both schemes from `porting-sdk/webhooks.md`:
4//!
5//! - **Scheme A** — RELAY/SWML/JSON: `lowercase_hex(HMAC-SHA1(key, url + raw_body))`
6//! - **Scheme B** — Compat/cXML form: `base64(HMAC-SHA1(key, url + sortedFormParams))`
7//!   with optional `bodySHA256` query-param fallback for JSON-on-compat-surface.
8//!
9//! Public API mirrors Python's `signalwire.core.security.webhook_validator`:
10//!
11//! - [`validate_webhook_signature`] — combined validator (tries A, then B).
12//! - [`validate_request`] — legacy `@signalwire/compatibility-api` drop-in
13//!   that accepts either a raw body string or pre-parsed form params.
14//!
15//! All comparisons go through [`subtle::ConstantTimeEq`] so the secret
16//! is not leaked through repeated requests. Inputs are taken as
17//! `&str` / borrowed strings — no allocation hidden behind the API.
18//!
19//! ```
20//! use signalwire::security::webhook::validate_webhook_signature;
21//!
22//! let key = "PSKtest1234567890abcdef";
23//! let url = "https://example.ngrok.io/webhook";
24//! let raw_body = r#"{"event":"call.state","params":{"call_id":"abc-123","state":"answered"}}"#;
25//! let sig = "c3c08c1fefaf9ee198a100d5906765a6f394bf0f";
26//! assert!(validate_webhook_signature(key, sig, url, raw_body).unwrap());
27//! ```
28//!

43/// Errors returned by the webhook validator.
44///
45/// `validate_webhook_signature` and `validate_request` return
46/// `Result<bool, WebhookError>` so callers can distinguish a legitimate
47/// "signature did not match" (`Ok(false)`) from a programming-error
48/// "you forgot the signing key" (`Err(WebhookError::MissingSigningKey)`).
49/// Per the spec, missing/empty *signature header* and malformed
50/// signatures return `Ok(false)` — those are not errors.
51#[derive(Debug, Clone, PartialEq, Eq)]
52pub enum WebhookError {
53    /// Caller passed an empty `signing_key`. This is a programming
54    /// error (the key is mandatory configuration), not a validation
55    /// failure — Python raises `ValueError` here, Node throws.
56    MissingSigningKey,
57}
58
59impl std::fmt::Display for WebhookError {
60    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
61        match self {
62            WebhookError::MissingSigningKey => write!(f, "signing_key is required"),
63        }
64    }
65}

69/// Argument shape for [`validate_request`]. Either a raw body string
70/// (which will run Scheme A first, then Scheme B with the form-parsed
71/// body) or a pre-parsed form-param list (which goes straight to
72/// Scheme B).
73///
74/// Repeated keys in the `Params` variant are supported by emitting a
75/// `Vec<String>` per key in submission order.
76#[derive(Debug, Clone)]
77pub enum ParamsOrBody {
78    /// Raw HTTP request body (UTF-8). The validator will sort form
79    /// params if it parses cleanly; for JSON bodies this branch
80    /// degrades to the empty-params Scheme B fallback.
81    Body(String),
82    /// Pre-parsed form params, list-of-(key, list-of-values) so
83    /// repeated keys keep their submission order.
84    Params(Vec<(String, Vec<String>)>),
85}

91/// Scheme-A digest: lowercase hex of HMAC-SHA1.
92fn hex_hmac_sha1(key: &str, message: &str) -> String {
93    let mut mac =
94        HmacSha1::new_from_slice(key.as_bytes()).expect("HMAC-SHA1 accepts any key length");
95    mac.update(message.as_bytes());
96    let digest = mac.finalize().into_bytes();
97    let mut out = String::with_capacity(digest.len() * 2);
98    for b in &digest {
99        let _ = write!(out, "{b:02x}");
100    }
101    out
102}

104/// Scheme-B digest: standard base64 of HMAC-SHA1.
105fn b64_hmac_sha1(key: &str, message: &str) -> String {
106    let mut mac =
107        HmacSha1::new_from_slice(key.as_bytes()).expect("HMAC-SHA1 accepts any key length");
108    mac.update(message.as_bytes());
109    let digest = mac.finalize().into_bytes();
110    BASE64_STANDARD.encode(digest)
111}

113/// Constant-time string compare. Uses `subtle::ConstantTimeEq`. Returns
114/// `false` if the lengths differ (length itself is non-secret —
115/// matches every other port's behavior).
116fn safe_eq(a: &str, b: &str) -> bool {
117    if a.len() != b.len() {
118        return false;
119    }
120    a.as_bytes().ct_eq(b.as_bytes()).into()
121}

123/// Concatenate form params per Scheme B rules:
124///
125/// - Sort by key, ASCII ascending.
126/// - For repeated keys: keep original submission order, emit
127///   `key + value` once per occurrence.
128/// - Stable sort preserves order within repeated keys.
129fn sorted_concat_params(params: &[(String, Vec<String>)]) -> String {
130    if params.is_empty() {
131        return String::new();
132    }
133    // Flatten: each (k, [v1, v2]) -> [(k,v1), (k,v2)] in submission order.
134    let mut flat: Vec<(&str, &str)> = Vec::with_capacity(params.len());
135    for (k, vs) in params {
136        for v in vs {
137            flat.push((k.as_str(), v.as_str()));
138        }
139    }
140    // Stable sort by key only.
141    flat.sort_by(|a, b| a.0.cmp(b.0));
142    let mut out = String::new();
143    for (k, v) in flat {
144        out.push_str(k);
145        out.push_str(v);
146    }
147    out
148}

150/// Parse an `application/x-www-form-urlencoded` body into ordered
151/// `(key, [values])` tuples. Repeated keys produce a single entry whose
152/// value-vector preserves submission order.
153fn parse_form_body(raw_body: &str) -> Vec<(String, Vec<String>)> {
154    if raw_body.is_empty() {
155        return Vec::new();
156    }
157    let mut order: Vec<String> = Vec::new();
158    let mut map: std::collections::BTreeMap<String, Vec<String>> =
159        std::collections::BTreeMap::new();
160    let mut insertion: Vec<(String, Vec<String>)> = Vec::new();
161
162    // Use url::form_urlencoded for percent-decoding; preserve original order.
163    for (k, v) in url::form_urlencoded::parse(raw_body.as_bytes()) {
164        let k = k.into_owned();
165        let v = v.into_owned();
166        if let Some(pos) = order.iter().position(|x| x == &k) {
167            // Append to the existing entry.
168            if let Some((_, vs)) = insertion.get_mut(pos) {
169                vs.push(v.clone());
170            }
171            map.entry(k).or_default().push(v);
172        } else {
173            order.push(k.clone());
174            insertion.push((k.clone(), vec![v.clone()]));
175            map.entry(k).or_default().push(v);
176        }
177    }
178    insertion
179}

181/// Return the URL variants to try for Scheme B port normalization.
182///
183/// SignalWire's backend signs some requests with the standard port
184/// included in the URL (`:443` for https, `:80` for http) and some
185/// without. The validator must try **both** forms.
186///
187/// We operate at the string level rather than going through `url::Url`
188/// because `Url::parse("https://x:443/")` normalises away the
189/// standard port — and the wire signature is over the literal URL bytes.
190///
191/// Rules:
192/// - If the URL has no port and scheme is http/https: emit input AND with-port.
193/// - If the URL has the scheme's standard port literal: emit input AND without-port.
194/// - Otherwise (non-standard explicit port, or non-http(s) scheme): emit input only.
195fn candidate_urls(url: &str) -> Vec<String> {
196    // Find scheme and the host portion in the raw string. We need the
197    // start index of the host (after `scheme://`) so we can splice
198    // `:<port>` in/out without disturbing the rest of the URL.
199    let Some(scheme_sep) = url.find("://") else {
200        return vec![url.to_string()];
201    };
202    let scheme = &url[..scheme_sep];
203    let standard_port = match scheme {
204        "http" => "80",
205        "https" => "443",
206        _ => return vec![url.to_string()],
207    };
208
209    let host_start = scheme_sep + 3;
210    // Host ends at the first '/', '?', '#', or end of string (whichever
211    // comes first). A `:` inside `[ipv6]` should not be confused with a
212    // port separator — handle that by skipping over `[...]` blocks.
213    let rest = &url[host_start..];
214    let mut host_end = rest.len();
215    let mut in_v6 = false;
216    for (i, ch) in rest.char_indices() {
217        match ch {
218            '[' => in_v6 = true,
219            ']' => in_v6 = false,
220            '/' | '?' | '#' if !in_v6 => {
221                host_end = i;
222                break;
223            }
224            _ => {}
225        }
226    }
227    let authority = &rest[..host_end];
228    let tail = &rest[host_end..];
229
230    // Find the port colon — must come after the closing `]` of an
231    // IPv6 host, or after a normal host.
232    let port_colon = if authority.starts_with('[') {
233        match authority.find(']') {
234            Some(close) => {
235                if close + 1 < authority.len() && authority.as_bytes().get(close + 1) == Some(&b':')
236                {
237                    Some(close + 1)
238                } else {
239                    None
240                }
241            }
242            None => None,
243        }
244    } else {
245        authority.rfind(':')
246    };
247
248    let (host_part, current_port): (&str, Option<&str>) = match port_colon {
249        Some(idx) => (&authority[..idx], Some(&authority[idx + 1..])),
250        None => (authority, None),
251    };
252
253    let mut candidates = vec![url.to_string()];
254
255    match current_port {
256        None => {
257            // No port → also try with the standard port.
258            let with_port = format!("{scheme}://{host_part}:{standard_port}{tail}");
259            if with_port != url {
260                candidates.push(with_port);
261            }
262        }
263        Some(p) if p == standard_port => {
264            // Standard port present → also try without it.
265            let without_port = format!("{scheme}://{host_part}{tail}");
266            if without_port != url {
267                candidates.push(without_port);
268            }
269        }
270        _ => { /* non-standard explicit port — only try as-is */ }
271    }
272    candidates
273}

275/// If the URL has `?bodySHA256=<hex>`, verify `sha256_hex(raw_body) == bodySHA256`.
276/// Returns `true` when the param is absent (no constraint) or matches.
277fn check_body_sha256(url: &str, raw_body: &str) -> bool {
278    let Ok(parsed) = Url::parse(url) else {
279        return true; // unparseable URL — let Scheme B fail naturally elsewhere
280    };
281    let expected = parsed
282        .query_pairs()
283        .find(|(k, _)| k == "bodySHA256")
284        .map(|(_, v)| v.into_owned());
285    let Some(expected) = expected else {
286        return true;
287    };
288    let mut hasher = Sha256::new();
289    hasher.update(raw_body.as_bytes());
290    let digest = hasher.finalize();
291    let mut hex = String::with_capacity(digest.len() * 2);
292    for b in &digest {
293        let _ = write!(hex, "{b:02x}");
294    }
295    safe_eq(&hex, &expected)
296}

302/// Validate a SignalWire webhook signature against both schemes.
303///
304/// # Arguments
305/// * `signing_key` — the customer's Signing Key from the SignalWire
306///   Dashboard. Empty / missing returns [`WebhookError::MissingSigningKey`]
307///   (this is a programming error, not a validation failure).
308/// * `signature` — the `X-SignalWire-Signature` (or `X-Twilio-Signature`)
309///   header value. Empty returns `Ok(false)` without raising.
310/// * `url` — the full URL SignalWire POSTed to (scheme, host, optional
311///   port, path, query). Must match exactly what the platform saw.
312/// * `raw_body` — the raw UTF-8 request body bytes, **before** any JSON
313///   or form parser consumes them. Re-serialization breaks the signature.
314///
315/// # Returns
316/// * `Ok(true)` if the signature matches Scheme A (hex JSON) or
317///   Scheme B (base64 form, with port-normalization variants and
318///   optional `bodySHA256` fallback).
319/// * `Ok(false)` if it does not match.
320/// * `Err(WebhookError::MissingSigningKey)` only when the key is missing.
321///
322/// # Errors
323///
324/// Returns `Err(WebhookError::MissingSigningKey)` when `signing_key` is
325/// empty — a caller/configuration error distinct from a signature
326/// mismatch. A wrong, malformed, or absent `signature` is *not* an
327/// error: it yields `Ok(false)`. This function only ever returns the
328/// one error variant.
329///
330/// All comparisons are constant-time via [`subtle::ConstantTimeEq`].
331pub fn validate_webhook_signature(
332    signing_key: &str,
333    signature: &str,
334    url: &str,
335    raw_body: &str,
336) -> Result<bool, WebhookError> {
337    if signing_key.is_empty() {
338        return Err(WebhookError::MissingSigningKey);
339    }
340    if signature.is_empty() {
341        return Ok(false);
342    }
343
344    // ------------------------------------------------------------------
345    // Scheme A — RELAY/SWML/JSON: hex(HMAC-SHA1(key, url + raw_body))
346    // ------------------------------------------------------------------
347    let mut a_input = String::with_capacity(url.len() + raw_body.len());
348    a_input.push_str(url);
349    a_input.push_str(raw_body);
350    let expected_a = hex_hmac_sha1(signing_key, &a_input);
351    if safe_eq(&expected_a, signature) {
352        return Ok(true);
353    }
354
355    // ------------------------------------------------------------------
356    // Scheme B — Compat/cXML form
357    // Two param shapes (parsed form params, then empty-for-JSON-on-compat),
358    // crossed with the URL port-normalization candidates.
359    // ------------------------------------------------------------------
360    let parsed_params = parse_form_body(raw_body);
361    let empty_params: Vec<(String, Vec<String>)> = Vec::new();
362    let param_shapes: [&[(String, Vec<String>)]; 2] = [&parsed_params, &empty_params];
363
364    for candidate_url in candidate_urls(url) {
365        for shape in &param_shapes {
366            let concat = sorted_concat_params(shape);
367            let mut b_input = String::with_capacity(candidate_url.len() + concat.len());
368            b_input.push_str(&candidate_url);
369            b_input.push_str(&concat);
370            let expected_b = b64_hmac_sha1(signing_key, &b_input);
371            if safe_eq(&expected_b, signature) {
372                // bodySHA256 (if present) must match too.
373                if check_body_sha256(&candidate_url, raw_body) {
374                    return Ok(true);
375                }
376                // Otherwise keep trying — caller may have signed a different
377                // shape that doesn't carry the bodySHA256 constraint.
378            }
379        }
380    }
381
382    Ok(false)
383}

385/// Legacy `@signalwire/compatibility-api` drop-in entry point.
386///
387/// If `params_or_raw_body` is [`ParamsOrBody::Body`], delegates to
388/// [`validate_webhook_signature`] (Scheme A then Scheme B with parsed form).
389///
390/// If it's [`ParamsOrBody::Params`], runs Scheme B directly with those
391/// pre-parsed form params (plus URL port normalization).
392///
393/// `bodySHA256` verification is skipped in the `Params` variant — there
394/// is no raw body to hash.
395///
396/// # Errors
397///
398/// Returns `Err(WebhookError::MissingSigningKey)` when `signing_key` is
399/// empty (in the `Body` variant this is surfaced by the delegated
400/// [`validate_webhook_signature`]). As above, an empty or non-matching
401/// `signature` is reported as `Ok(false)`, not an error.
402pub fn validate_request(
403    signing_key: &str,
404    signature: &str,
405    url: &str,
406    params_or_raw_body: &ParamsOrBody,
407) -> Result<bool, WebhookError> {
408    if signing_key.is_empty() {
409        return Err(WebhookError::MissingSigningKey);
410    }
411    if signature.is_empty() {
412        return Ok(false);
413    }
414
415    match params_or_raw_body {
416        ParamsOrBody::Body(b) => validate_webhook_signature(signing_key, signature, url, b),
417        ParamsOrBody::Params(params) => {
418            let concat = sorted_concat_params(params);
419            for candidate_url in candidate_urls(url) {
420                let mut b_input = String::with_capacity(candidate_url.len() + concat.len());
421                b_input.push_str(&candidate_url);
422                b_input.push_str(&concat);
423                let expected_b = b64_hmac_sha1(signing_key, &b_input);
424                if safe_eq(&expected_b, signature) {
425                    return Ok(true);
426                }
427            }
428            Ok(false)
429        }
430    }
431}

688    #[test]
689    fn safe_eq_uses_subtle_constant_time() {
690        // The validator must not branch on signature contents. We assert
691        // safe_eq is exposed as a function (not inlined macros) and that
692        // a well-known byte-equal pair returns true while length-mismatch
693        // returns false. The deeper "is it actually constant-time" is
694        // delegated to the `subtle` crate's audited implementation.
695        assert!(safe_eq("abc", "abc"));
696        assert!(!safe_eq("abc", "abd"));
697        assert!(!safe_eq("abc", "abcd"));
698        assert!(!safe_eq("", "x"));
699    }
700
701    #[test]
702    fn validator_source_uses_subtle_constant_time_eq() {
703        // Source-level audit: the spec mandates constant-time compare.
704        // We assert the module imports `subtle::ConstantTimeEq` and
705        // that the production code path (everything before the
706        // `#[cfg(test)] mod tests` block) does not compare a digest
707        // with plain `==`.
708        let full_src = include_str!("webhook.rs");
709        assert!(
710            full_src.contains("subtle::ConstantTimeEq"),
711            "webhook validator must import subtle::ConstantTimeEq"
712        );
713        assert!(
714            full_src.contains(".ct_eq("),
715            "safe_eq must call .ct_eq(...) (the constant-time API)"
716        );
717
718        // Strip the test module so we audit only production code.
719        let prod_src = match full_src.find("#[cfg(test)]") {
720            Some(idx) => &full_src[..idx],
721            None => full_src,
722        };
723
724        // Negative: production code must not do `expected == signature`
725        // anywhere — that's the leaky pattern the constant-time compare
726        // exists to replace.
727        for forbidden in [
728            "expected_a == signature",
729            "expected_b == signature",
730            "expected == signature",
731        ] {
732            assert!(
733                !prod_src.contains(forbidden),
734                "production code must not use plain == on digest: found {forbidden:?}"
735            );
736        }
737    }
