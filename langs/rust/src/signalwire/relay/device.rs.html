1//! Typed RELAY device object.
2//!
3//! The `{ "type": <string>, "params": <object> }` device object recurs as a
4//! raw JSON blob across `calling.dial` / `calling.connect` / `calling.refer` /
5//! `calling.tap` (and inside the serial/parallel device *matrix*
6//! `[[device, …], …]`). [`Device`] types the **shape** of that blob while
7//! deliberately keeping `type` a plain [`String`]: the discriminant
8//! (`phone` / `sip` / `webrtc` / `rtp` / …) is **not** schema-enumerated in
9//! `relay-protocol/calling.{dial,connect,refer,tap}.params.json` (each pins
10//! only `{type: string, params}`), so an enum would be inventing a closed set
11//! the contract does not declare.
12//!
13//! This is **additive**: every existing raw-`serde_json::Value` entry point
14//! (`Call::connect`, `Call::refer_call`, `Call::tap`,
15//! `Client::dial_blocking`) is unchanged. [`Device::to_value`] serialises to
16//! the **byte-identical** wire object the hand-written `json!({...})` produces
17//! (field order `type` then `params`, `params` always an object), so a typed
18//! device drops straight into any of those raw-`Value` parameters.
19
20use serde_json::{Map, Value};
21
22/// A typed RELAY device descriptor (`{type, params}`).
23///
24/// `type` is the device-kind discriminant (kept as a `String` — the wire
25/// contract does not enumerate it). `params` is the device-specific
26/// parameter object (e.g. `{to_number, from_number}` for a `phone`).
27///
28/// Build one with [`Device::new`] (raw params) or [`Device::phone`] /
29/// [`Device::sip`] convenience constructors, then hand its
30/// [`to_value`](Device::to_value) to any device-taking call:
31///
32/// ```ignore
33/// let dev = Device::phone("+15551112222", "+15553334444");
34/// // single parallel leg = one inner list
35/// let devices = Device::matrix(&[&[dev]]);
36/// client.dial_blocking(devices, None, None, timeout);
37/// ```
38#[derive(Debug, Clone, PartialEq)]
39pub struct Device {
40    /// Device kind discriminant (`phone`, `sip`, `webrtc`, `rtp`, …). A
41    /// plain string: the wire schema does not enumerate it.
42    pub device_type: String,
43    /// Device-specific parameters object.
44    pub params: Value,
45}
46
47impl Device {
48    /// Build a device from its `type` discriminant and a `params` object.
49    ///
50    /// A non-object `params` (or [`Value::Null`]) is normalised to an empty
51    /// object on the wire, matching the hand-written `"params": {}` form.
52    #[must_use]
53    pub fn new(device_type: impl Into<String>, params: Value) -> Self {
54        Device {
55            device_type: device_type.into(),
56            params,
57        }
58    }
59
60    /// A `phone` device with `to_number` / `from_number` params.
61    ///
62    /// Emits `{"type":"phone","params":{"to_number":…,"from_number":…}}` —
63    /// the shape `Client::dial_blocking` and `Call::connect` expect for PSTN
64    /// legs.
65    #[must_use]
66    pub fn phone(to_number: impl Into<String>, from_number: impl Into<String>) -> Self {
67        Device::new(
68            "phone",
69            Value::Object({
70                let mut m = Map::new();
71                m.insert("to_number".to_string(), Value::String(to_number.into()));
72                m.insert("from_number".to_string(), Value::String(from_number.into()));
73                m
74            }),
75        )
76    }
77
78    /// A `sip` device addressed to a SIP URI.
79    ///
80    /// Emits `{"type":"sip","params":{"to":…,"from":…}}`.
81    #[must_use]
82    pub fn sip(to: impl Into<String>, from: impl Into<String>) -> Self {
83        Device::new(
84            "sip",
85            Value::Object({
86                let mut m = Map::new();
87                m.insert("to".to_string(), Value::String(to.into()));
88                m.insert("from".to_string(), Value::String(from.into()));
89                m
90            }),
91        )
92    }
93
94    /// Serialise to the wire device object: `{"type": …, "params": {…}}`.
95    ///
96    /// Field order is `type` then `params` and `params` is always an object,
97    /// so the output is byte-identical to the hand-written
98    /// `json!({"type": t, "params": p})` that every existing device call
99    /// site uses. (`serde_json` is built with `preserve_order`, so insertion
100    /// order is the serialised order.)
101    #[must_use]
102    pub fn to_value(&self) -> Value {
103        let params = match &self.params {
104            Value::Object(_) => self.params.clone(),
105            _ => Value::Object(Map::new()),
106        };
107        let mut obj = Map::new();
108        obj.insert("type".to_string(), Value::String(self.device_type.clone()));
109        obj.insert("params".to_string(), params);
110        Value::Object(obj)
111    }
112
113    /// Build the serial/parallel device **matrix** (`[[device, …], …]`) that
114    /// `dial` / `connect` take, from rows of devices.
115    ///
116    /// Each inner slice is one *serial* attempt; the outer slice runs its
117    /// rows in *parallel*. `Device::matrix(&[&[a, b]])` = try `a` then `b`
118    /// in one parallel leg; `Device::matrix(&[&[a], &[b]])` = ring `a` and
119    /// `b` in parallel.
120    #[must_use]
121    pub fn matrix(rows: &[&[Device]]) -> Value {
122        Value::Array(
123            rows.iter()
124                .map(|row| Value::Array(row.iter().map(Device::to_value).collect()))
125                .collect(),
126        )
127    }
128}

134#[cfg(test)]
135mod tests {
136    use super::*;
137    use serde_json::json;
138
139    #[test]
140    fn to_value_is_byte_identical_to_handwritten() {
141        let dev = Device::phone("+15551112222", "+15553334444");
142        let hand = json!({
143            "type": "phone",
144            "params": {"to_number": "+15551112222", "from_number": "+15553334444"},
145        });
146        // Structural equality.
147        assert_eq!(dev.to_value(), hand);
148        // Byte-identical serialisation (field order preserved).
149        assert_eq!(
150            serde_json::to_string(&dev.to_value()).unwrap(),
151            serde_json::to_string(&hand).unwrap(),
152        );
153        assert_eq!(
154            serde_json::to_string(&dev.to_value()).unwrap(),
155            r#"{"type":"phone","params":{"to_number":"+15551112222","from_number":"+15553334444"}}"#,
156        );
157    }
158
159    #[test]
160    fn sip_shape() {
161        let dev = Device::sip("sip:bob@example.com", "sip:alice@example.com");
162        assert_eq!(
163            dev.to_value(),
164            json!({
165                "type": "sip",
166                "params": {"to": "sip:bob@example.com", "from": "sip:alice@example.com"},
167            }),
168        );
169    }
170
171    #[test]
172    fn new_with_raw_params_round_trips() {
173        let dev = Device::new("rtp", json!({"addr": "203.0.113.1", "port": 4000}));
174        let hand = json!({"type": "rtp", "params": {"addr": "203.0.113.1", "port": 4000}});
175        assert_eq!(dev.to_value(), hand);
176        assert_eq!(
177            serde_json::to_string(&dev.to_value()).unwrap(),
178            serde_json::to_string(&hand).unwrap(),
179        );
180    }
181
182    #[test]
183    fn non_object_params_normalised_to_empty_object() {
184        // A device with no params must still emit "params": {} (the form the
185        // tap/connect schemas accept and existing call sites hand-write).
186        let dev = Device::new("phone", Value::Null);
187        assert_eq!(dev.to_value(), json!({"type": "phone", "params": {}}));
188    }
189
190    #[test]
191    fn matrix_single_parallel_leg() {
192        let a = Device::phone("+1", "+2");
193        let m = Device::matrix(&[std::slice::from_ref(&a)]);
194        let hand = json!([[{"type": "phone", "params": {"to_number": "+1", "from_number": "+2"}}]]);
195        assert_eq!(m, hand);
196        assert_eq!(
197            serde_json::to_string(&m).unwrap(),
198            serde_json::to_string(&hand).unwrap(),
199        );
200    }
201
202    #[test]
203    fn matrix_serial_and_parallel() {
204        let a = Device::phone("+1", "+2");
205        let b = Device::sip("sip:b", "sip:a");
206        // one parallel leg, two serial devices
207        let serial = Device::matrix(&[&[a.clone(), b.clone()]]);
208        assert_eq!(serial.as_array().unwrap().len(), 1);
209        assert_eq!(serial[0].as_array().unwrap().len(), 2);
210        // two parallel legs, one device each
211        let parallel = Device::matrix(&[&[a], &[b]]);
212        assert_eq!(parallel.as_array().unwrap().len(), 2);
213        assert_eq!(parallel[0].as_array().unwrap().len(), 1);
214    }
