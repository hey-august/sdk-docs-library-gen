1//! Typed error for the RELAY client surface.
2//!
3//! The REST layer already ships a proper error enum
4//! ([`SignalWireRestError`](crate::rest::SignalWireRestError)); this is the
5//! RELAY-side analogue. Before this, the relay client returned
6//! `Result<_, String>` everywhere — a stringly-typed failure channel that
7//! forces callers to pattern-match on message *text* to react differently to
8//! (say) an auth rejection vs. a transport drop vs. a timeout. [`RelayError`]
9//! makes the failure modes a closed, inspectable set, exactly like the REST
10//! exemplar.
11//!
12//! Design notes (idiomatic Rust, matching the REST error's bar):
13//! - The data each failure carries lives **in the variant**, so callers react
14//!   by `match`ing rather than calling accessor getters. That keeps the public
15//!   surface to the type + its trait impls (`Display` / [`std::error::Error`])
16//!   and means the audit sees no new methods.
17//! - `#[non_exhaustive]` because the failure set mirrors transport/server
18//!   conditions that can grow (new RELAY error classes) without that being a
19//!   breaking change — downstream `match` must carry a wildcard.
20//! - Every variant's [`Display`] is human-actionable and preserves the same
21//!   context the old `format!("…")` strings carried, so log output is no worse.
22
23use std::fmt;

25/// Something went wrong talking to the RELAY WebSocket service.
26///
27/// Returned by the connection-lifecycle and blocking-RPC methods on
28/// [`Client`](crate::relay::Client) (e.g. `connect`, `authenticate_blocking`,
29/// `execute_blocking`, `dial_blocking`, `send_message_blocking`).
30#[derive(Debug, Clone, PartialEq, Eq)]
31#[non_exhaustive]
32pub enum RelayError {
33    /// A required environment variable was missing when building the client
34    /// from the environment (`from_env`). Carries the variable name.
35    MissingEnv {
36        /// The environment variable that was not set.
37        var: String,
38    },
39
40    /// The underlying WebSocket transport failed to connect, upgrade, read, or
41    /// write. Carries a human description of the transport-level cause.
42    Transport {
43        /// What was being attempted (e.g. the target URL or operation).
44        context: String,
45        /// The transport library's error rendered to text.
46        source: String,
47    },
48
49    /// The RELAY server rejected the `signalwire.connect` authentication
50    /// handshake. Carries the server-reported reason.
51    Auth {
52        /// The server's error message (or a default if none was supplied).
53        message: String,
54    },
55
56    /// The server returned a JSON-RPC error for a request (e.g. a failed
57    /// `messaging.send` / `calling.dial`). Carries the inner method and the
58    /// server's error message.
59    Rpc {
60        /// The RELAY method that failed (e.g. `messaging.send`).
61        method: String,
62        /// The server's error message.
63        message: String,
64    },
65
66    /// A blocking call did not receive its response within the deadline.
67    /// Carries what was being awaited so the log line is actionable.
68    Timeout {
69        /// What timed out (e.g. `signalwire.connect` or a dial tag).
70        what: String,
71    },
72
73    /// A `dial` completed with a `failed` state or never produced an answer
74    /// before the dial deadline. Carries the reason / tag.
75    DialFailed {
76        /// The dial reason or tag context.
77        reason: String,
78    },
79
80    /// A required argument was invalid (e.g. `send_message` with neither body
81    /// nor media). Carries the validation message.
82    InvalidArgument {
83        /// The validation failure description.
84        message: String,
85    },
86}
87
88impl RelayError {
89    /// Build a [`RelayError::Transport`] from a context and any `Display` cause.
90    /// Convenience for the `map_err(|e| …)` call sites in the client.
91    pub fn transport(context: impl Into<String>, source: impl fmt::Display) -> Self {
92        RelayError::Transport {
93            context: context.into(),
94            source: source.to_string(),
95        }
96    }
97
98    /// Build a [`RelayError::MissingEnv`] for the named variable.
99    pub fn missing_env(var: impl Into<String>) -> Self {
100        RelayError::MissingEnv { var: var.into() }
101    }
102}
103
104impl fmt::Display for RelayError {
105    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
106        match self {
107            RelayError::MissingEnv { var } => {
108                write!(f, "RelayError: {var} not set")
109            }
110            RelayError::Transport { context, source } => {
111                write!(f, "RelayError: transport ({context}): {source}")
112            }
113            RelayError::Auth { message } => {
114                write!(f, "RelayError: auth error: {message}")
115            }
116            RelayError::Rpc { method, message } => {
117                write!(f, "RelayError: {method} failed: {message}")
118            }
119            RelayError::Timeout { what } => {
120                write!(f, "RelayError: timed out waiting for {what}")
121            }
122            RelayError::DialFailed { reason } => {
123                write!(f, "RelayError: dial failed: {reason}")
124            }
125            RelayError::InvalidArgument { message } => {
126                write!(f, "RelayError: invalid argument: {message}")
127            }
128        }
129    }
130}

134#[cfg(test)]
135mod tests {
136    use super::*;
137
138    #[test]
139    fn test_variants_carry_their_data() {
140        let e = RelayError::missing_env("SIGNALWIRE_PROJECT_ID");
141        assert_eq!(
142            e,
143            RelayError::MissingEnv {
144                var: "SIGNALWIRE_PROJECT_ID".into()
145            }
146        );
147
148        let t = RelayError::transport("WS connect to wss://x", "handshake refused");
149        match t {
150            RelayError::Transport { context, source } => {
151                assert!(context.contains("wss://x"));
152                assert_eq!(source, "handshake refused");
153            }
154            _ => panic!("wrong variant"),
155        }
156    }
157
158    #[test]
159    fn test_display_is_actionable() {
160        assert!(
161            RelayError::Auth {
162                message: "bad token".into()
163            }
164            .to_string()
165            .contains("bad token")
166        );
167        assert!(
168            RelayError::Timeout {
169                what: "signalwire.connect".into()
170            }
171            .to_string()
172            .contains("signalwire.connect")
173        );
174        assert!(
175            RelayError::Rpc {
176                method: "messaging.send".into(),
177                message: "rejected".into()
178            }
179            .to_string()
180            .contains("messaging.send")
181        );
182    }
183
184    #[test]
185    fn test_is_std_error() {
186        let e = RelayError::DialFailed {
187            reason: "no answer".into(),
188        };
189        let _: &dyn std::error::Error = &e;
190    }
