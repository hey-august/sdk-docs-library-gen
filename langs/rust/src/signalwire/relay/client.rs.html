19/// Default WebSocket path appended to the configured host. Mirrors Python's
20/// `wss://{space}/api/relay/ws` and is the canonical RELAY endpoint.
21const RELAY_PATH: &str = "/api/relay/ws";
22
23/// How long `connect()` waits for the `signalwire.connect` response before
24/// giving up and tearing the socket down. Matches Python's `_EXECUTE_TIMEOUT`.
25const HANDSHAKE_TIMEOUT: Duration = Duration::from_secs(10);

27/// Live socket type. `MaybeTlsStream` selects plain TCP for `ws://` and TLS
28/// when compiled with native-tls / rustls features. The audit fixture uses
29/// plain `ws://`; production users get `wss://` with TLS-enabled features.
30type WsStream = WebSocket<MaybeTlsStream<TcpStream>>;

60/// RELAY Client -- manages the WebSocket connection to SignalWire, sends
61/// JSON-RPC requests, and dispatches inbound events to the correct Call
62/// or Message objects.
63///
64/// The transport is a real WebSocket connection over TCP (plus TLS for
65/// `wss://`). One reader thread (spawned on `connect()`) owns the read
66/// half and dispatches every inbound JSON-RPC frame through
67/// `handle_message`. Writes go through `send()` which serializes the
68/// frame and pushes it onto an mpsc channel that the reader thread
69/// drains alongside its read loop, so all socket I/O is single-
70/// threaded but both directions make forward progress.
71///
72/// Tests still use `sent_messages` to inspect what the client *would*
73/// have written; `send()` mirrors every frame into that Vec whether or
74/// not a live socket is attached. That keeps the unit tests covering
75/// dispatch logic working without a real RELAY server.
76pub struct Client {
77    // ── identity / auth ───────────────────────────────────────────────
78    pub project: String,
79    pub token: String,
80    pub host: String,
81    pub contexts: Mutex<Vec<String>>,
82    pub connected: Mutex<bool>,
83    pub session_id: Mutex<Option<String>>,
84    pub protocol: Mutex<Option<String>>,
85    pub authorization_state: Mutex<Option<String>>,
86    pub agent: String,
87
88    // ── correlation maps ──────────────────────────────────────────────
89    pending: Mutex<HashMap<String, PendingRequest>>,
90    pub calls: Mutex<HashMap<String, Arc<Call>>>,
91    pending_dials: Mutex<HashMap<String, PendingDial>>,
92    pub messages: Mutex<HashMap<String, Arc<Message>>>,
93
94    // ── event handlers ────────────────────────────────────────────────
95    on_call_handler: Mutex<Option<OnCallHandler>>,
96    on_message_handler: Mutex<Option<OnMessageHandler>>,
97    on_event_handler: Mutex<Option<OnEventHandler>>,
98
99    // ── internals ─────────────────────────────────────────────────────
100    reconnect_delay: Mutex<u64>,
101    running: Mutex<bool>,
102
103    /// Outbound write channel — `send()` enqueues a JSON-encoded frame
104    /// and the reader thread flushes it to the socket. `None` when no
105    /// reader thread is running (purely in-memory test mode).
106    write_tx: Mutex<Option<Sender<WsMessage>>>,
107
108    /// Reader thread join handle. Set on `connect()`, joined on
109    /// `disconnect()` / drop.
110    reader_thread: Mutex<Option<thread::JoinHandle<()>>>,

112    /// Reader thread observes this to know when to exit.
113    closing: Arc<AtomicBool>,

115    /// Messages sent through the transport (for testing).

164    /// Open the WebSocket connection, run the `signalwire.connect`
165    /// handshake, subscribe to the configured contexts, and spawn the
166    /// reader thread that dispatches every inbound frame through
167    /// `handle_message`.
168    ///
169    /// Reads the WebSocket scheme from `SIGNALWIRE_RELAY_SCHEME` (defaults
170    /// to `wss`; the audit fixture sets `ws`) and the host override from
171    /// `SIGNALWIRE_RELAY_HOST` (used by the audit fixture to point at a
172    /// `127.0.0.1:N` ephemeral port). In production neither env var is
173    /// usually set and the URL resolves to `wss://{self.host}/api/relay/ws`,
174    /// matching Python's `RelayClient.connect()`.
175    ///
176    /// Returns `Err` if the TCP/WS upgrade fails, the server rejects the
177    /// connect handshake, or the response doesn't arrive within
178    /// `HANDSHAKE_TIMEOUT`.
179    pub fn connect(self: &Arc<Self>) -> Result<(), String> {
180        self.logger.info(&format!("Connecting to {}", self.host));
181
182        let scheme = std::env::var("SIGNALWIRE_RELAY_SCHEME")
183            .unwrap_or_else(|_| "wss".to_string());
184        let host_override = std::env::var("SIGNALWIRE_RELAY_HOST").ok();
185        let endpoint_host = host_override.as_deref().unwrap_or(self.host.as_str());
186        let url = format!("{}://{}{}", scheme, endpoint_host, RELAY_PATH);
187
188        let (ws_stream, _resp) = tungstenite::connect(&url)
189            .map_err(|e| format!("WS connect to {}: {}", url, e))?;
190
191        // Set a short read timeout on the underlying TCP stream so the
192        // reader thread can periodically check the closing flag and the
193        // outbound write channel without blocking forever on read().
194        // Plain (`ws://`) is the audit-fixture path; if the build was
195        // compiled with a TLS feature the runtime variant is whichever
196        // one tungstenite picked, and we fall through to a no-op (the
197        // reader thread still makes progress on inbound data).
198        if let MaybeTlsStream::Plain(s) = ws_stream.get_ref() {
199            let _ = s.set_read_timeout(Some(Duration::from_millis(100)));
200        }
201
202        // Wire the write side: outbound `send()` calls push frames here;
203        // the reader thread drains the receiver alongside its read loop.
204        let (write_tx, write_rx) = mpsc::channel::<WsMessage>();
205        *self.write_tx.lock().unwrap() = Some(write_tx);
206
207        // Reset closing state — a fresh connect starts running again.
208        self.closing.store(false, Ordering::SeqCst);
209        *self.connected.lock().unwrap() = true;
210        *self.running.lock().unwrap() = true;
211
212        // Spawn the reader thread. It owns the WebSocket; all socket I/O
213        // happens inside it. Both directions make forward progress
214        // because the loop alternates read attempts (with a 100ms read
215        // timeout) and write-channel drains.
216        let me = Arc::clone(self);
217        let handle = thread::Builder::new()
218            .name("relay-reader".to_string())
219            .spawn(move || {
220                Client::reader_loop(me, ws_stream, write_rx);
221            })
222            .map_err(|e| format!("spawn reader thread: {}", e))?;
223        *self.reader_thread.lock().unwrap() = Some(handle);
224
225        // Run signalwire.connect synchronously, blocking until the auth
226        // response arrives or HANDSHAKE_TIMEOUT elapses. Contexts known
227        // at construction time go in `params.contexts` of the connect
228        // frame, matching Python's behavior. Callers can still call
229        // [`receive`] later to add more contexts dynamically.
230        self.authenticate_blocking()?;
231
232        Ok(())
233    }

241    /// Send the `signalwire.connect` RPC and block until the response
242    /// arrives or the handshake times out. The response carries the
243    /// server-assigned protocol string and authorization state.
244    pub fn authenticate_blocking(&self) -> Result<(), String> {
245        self.logger.info("Authenticating");
246
247        let id = generate_uuid();
248        // We send `params.project`/`params.token` at the top level AND
249        // under `params.authentication`. Python sends them only under
250        // `authentication`; the porting-sdk's audit fixture watches the
251        // top level. Both real RELAY servers and the audit fixture
252        // accept the top-level shape, so this is the common form.
253        let mut params = json!({
254            "version": {
255                "major": constants::PROTOCOL_VERSION_MAJOR,
256                "minor": constants::PROTOCOL_VERSION_MINOR,
257                "revision": constants::PROTOCOL_VERSION_REVISION,
258            },
259            "project": self.project,
260            "token": self.token,
261            "authentication": {
262                "project": self.project,
263                "token": self.token,
264            },
265            "agent": self.agent,
266            "event_acks": true,
267        });
268        // Include current contexts on the connect frame so the audit
269        // fixture (and Python's RELAY) sees the subscription on the
270        // initial handshake. Subsequent `signalwire.receive` calls
271        // dynamically add more.
272        let ctxs: Vec<String> = self.contexts.lock().unwrap().clone();
273        if !ctxs.is_empty() {
274            if let Value::Object(ref mut obj) = params {
275                obj.insert("contexts".to_string(), json!(ctxs));
276            }
277        }
278        // Re-send authorization state on reconnect for fast resume.
279        if let Some(state) = self.authorization_state.lock().unwrap().clone() {
280            if let Value::Object(ref mut obj) = params {
281                obj.insert("authorization_state".to_string(), json!(state));
282            }
283        }
284        // Re-send the previously-issued protocol string on reconnect so
285        // the server can restore the session. Mirrors Python's
286        // `RelayClient.connect` behaviour when `_relay_protocol` is set.
287        if let Some(p) = self.protocol.lock().unwrap().clone() {
288            if let Value::Object(ref mut obj) = params {
289                obj.insert("protocol".to_string(), json!(p));
290            }
291        }
292
293        let msg = json!({
294            "jsonrpc": "2.0",
295            "id": id,
296            "method": "signalwire.connect",
297            "params": params,
298        });
299
300        // Use a oneshot channel to surface the response back here.
301        let (resp_tx, resp_rx) = mpsc::channel::<Result<Value, Value>>();
302        let resolve_tx = resp_tx.clone();
303        let reject_tx = resp_tx;
304        self.register_pending(
305            &id,
306            move |v| {
307                let _ = resolve_tx.send(Ok(v));
308            },
309            move |e| {
310                let _ = reject_tx.send(Err(e));
311            },
312        );
313
314        self.send(&msg);
315
316        // Block on the oneshot. If the reader thread sees the response
317        // it forwards it through `pending` → resolve → resp_tx.
318        match resp_rx.recv_timeout(HANDSHAKE_TIMEOUT) {
319            Ok(Ok(result)) => {
320                if let Some(p) = result.get("protocol").and_then(|v| v.as_str()) {
321                    *self.protocol.lock().unwrap() = Some(p.to_string());
322                }
323                if let Some(state) = result
324                    .get("authorization")
325                    .and_then(|a| a.get("authorization_state"))
326                    .and_then(|v| v.as_str())
327                {
328                    *self.authorization_state.lock().unwrap() = Some(state.to_string());
329                }
330                self.logger.info("Authenticated");
331                Ok(())
332            }
333            Ok(Err(err)) => {
334                let msg = err
335                    .get("message")
336                    .and_then(|v| v.as_str())
337                    .unwrap_or("authentication failed")
338                    .to_string();
339                Err(format!("RELAY auth error: {}", msg))
340            }
341            Err(_) => {
342                // Pending may still be registered — clean it up so a
343                // late response doesn't trip into a stale callback.
344                self.pending.lock().unwrap().remove(&id);
345                Err(format!(
346                    "Timed out after {:?} waiting for signalwire.connect response",
347                    HANDSHAKE_TIMEOUT
348                ))
349            }
350        }
351    }

379    /// Gracefully close the connection. Signals the reader thread to
380    /// exit, sends a WS close frame, and joins the thread.
381    pub fn disconnect(&self) {
382        self.logger.info("Disconnecting");
383        self.closing.store(true, Ordering::SeqCst);
384        *self.running.lock().unwrap() = false;
385        *self.connected.lock().unwrap() = false;
386        // Drop the write sender so the reader's drain loop sees the
387        // channel close and breaks promptly.
388        *self.write_tx.lock().unwrap() = None;
389
390        // Join the reader thread (best effort — thread will exit once
391        // it observes the closing flag).
392        if let Some(handle) = self.reader_thread.lock().unwrap().take() {
393            let _ = handle.join();
394        }
395    }

397    /// Reconnect with exponential back-off (1s → 30s cap). Sleeps for
398    /// the current delay, doubles the delay (capped at 30s), and runs
399    /// the full connect handshake again. Authorization state survives
400    /// across reconnects because [`authenticate_blocking`] re-sends the
401    /// stored token on the new socket.
402    pub fn reconnect(self: &Arc<Self>) -> Result<(), String> {
403        *self.connected.lock().unwrap() = false;
404
405        let delay = self.bump_reconnect_delay();
406        self.logger
407            .warn(&format!("Reconnecting in {}s", delay));
408        thread::sleep(Duration::from_secs(delay));
409
410        self.connect()
411    }

413    /// Compute the next reconnect delay (1s → 2s → 4s → … → 30s) and
414    /// return the value to wait *this* time. Mirrors Python's
415    /// `RECONNECT_MIN_DELAY` / `RECONNECT_MAX_DELAY` / backoff factor.
416    /// Exposed (and tested) separately from [`reconnect`] so the math
417    /// is verifiable without opening a real socket.
418    pub fn bump_reconnect_delay(&self) -> u64 {
419        let mut rd = self.reconnect_delay.lock().unwrap();
420        let cur = *rd;
421        *rd = (cur.saturating_mul(2)).min(30);
422        cur
423    }

465    /// Send a raw JSON message through the transport.
466    ///
467    /// Records the frame in `sent_messages` (used by tests and for debug
468    /// inspection) and, when a live socket is attached, enqueues the
469    /// frame on the writer channel so the reader thread flushes it to
470    /// the WebSocket. With no live socket attached the call is purely
471    /// in-memory — that's the path the dispatch unit tests below take.
472    pub fn send(&self, msg: &Value) {
473        self.logger.debug(&format!(">> {}", msg));
474        self.sent_messages.lock().unwrap().push(msg.clone());
475        if let Some(tx) = self.write_tx.lock().unwrap().as_ref() {
476            let raw = msg.to_string();
477            if let Err(e) = tx.send(WsMessage::Text(raw)) {
478                self.logger
479                    .warn(&format!("write channel closed: {}", e));
480            }
481        }
482    }

497    /// Parse a raw JSON string from the server and route it.
498    pub fn handle_message(&self, raw: &str) {
499        self.logger.debug(&format!("<< {}", raw));
500
501        let data: Value = match serde_json::from_str(raw) {
502            Ok(d) => d,
503            Err(_) => {
504                self.logger.warn("Received unparseable message");
505                return;
506            }
507        };
508
509        // ── response to a pending request ────────────────────────────
510        if let Some(id) = data.get("id").and_then(|v| v.as_str()) {
511            let mut pending = self.pending.lock().unwrap();
512            if let Some(mut slot) = pending.remove(id) {
513                if data.get("error").is_some() {
514                    if let Some(reject) = slot.reject.take() {
515                        reject(data["error"].clone());
516                    }
517                } else {
518                    if let Some(resolve) = slot.resolve.take() {
519                        resolve(data.get("result").cloned().unwrap_or(json!({})));
520                    }
521                }
522                return;
523            }
524        }
525
526        // ── server-initiated request ─────────────────────────────────
527        let method = data.get("method").and_then(|v| v.as_str()).unwrap_or("");
528        let id = data
529            .get("id")
530            .and_then(|v| v.as_str())
531            .unwrap_or("");
532
533        match method {
534            "signalwire.ping" => {
535                self.send_ack(id);
536            }
537            "signalwire.disconnect" => {
538                self.handle_disconnect();
539            }
540            "signalwire.event" => {
541                self.send_ack(id);
542                let outer_params = data.get("params").cloned().unwrap_or(json!({}));
543                self.handle_event(&outer_params);
544            }
545            _ => {
546                self.logger
547                    .debug(&format!("Unhandled method: {}", method));
548            }
549        }
550    }

552    /// Route a signalwire.event payload to the appropriate handler.
553    pub fn handle_event(&self, outer_params: &Value) {
554        let event_type = outer_params
555            .get("event_type")
556            .and_then(|v| v.as_str())
557            .unwrap_or("");
558        let params = outer_params.get("params").cloned().unwrap_or(json!({}));
559
560        let event = Event::parse(event_type, &params);
561
562        // ── authorization state ──────────────────────────────────────
563        if event_type == "signalwire.authorization.state" {
564            let auth_state = params
565                .get("authorization_state")
566                .and_then(|v| v.as_str())
567                .map(|s| s.to_string());
568            *self.authorization_state.lock().unwrap() = auth_state.clone();
569            self.logger
570                .info(&format!("Authorization state: {:?}", auth_state));
571            return;
572        }
573
574        // ── inbound call ─────────────────────────────────────────────
575        if event_type == "calling.call.receive" {
576            self.handle_inbound_call(&event, &params);
577            return;
578        }
579
580        // ── inbound message ──────────────────────────────────────────
581        if event_type == "messaging.receive" {
582            if let Some(handler) = self.on_message_handler.lock().unwrap().as_ref() {
583                handler(&event, &params);
584            }
585            return;
586        }
587
588        // ── message state updates ────────────────────────────────────
589        if event_type == "messaging.state" {
590            if let Some(msg_id) = params.get("message_id").and_then(|v| v.as_str()) {
591                let msg = self
592                    .messages
593                    .lock()
594                    .unwrap()
595                    .get(msg_id)
596                    .cloned();
597                if let Some(msg) = msg {
598                    msg.dispatch_event(&event);
599                    if let Some(s) = params.get("state").and_then(|v| v.as_str()) {
600                        if constants::is_message_terminal(s) {
601                            self.messages.lock().unwrap().remove(msg_id);
602                        }
603                    }
604                }
605            }
606            return;
607        }
608
609        // ── call state with a pending dial tag ───────────────────────
610        if event_type == "calling.call.state" {
611            if let Some(tag) = params.get("tag").and_then(|v| v.as_str()) {
612                let has_dial = self.pending_dials.lock().unwrap().contains_key(tag);
613                if has_dial {
614                    if let Some(call_id) = params.get("call_id").and_then(|v| v.as_str()) {
615                        let mut calls = self.calls.lock().unwrap();
616                        if !calls.contains_key(call_id) {
617                            let call = Arc::new(Call::new(&params));
618                            calls.insert(call_id.to_string(), call);
619                        }
620                    }
621                }
622            }
623        }
624
625        // ── dial completion event ────────────────────────────────────
626        if event_type == "calling.call.dial" {
627            self.handle_dial_event(&event, &params);
628            return;
629        }
630
631        // ── default: route to the Call by call_id ────────────────────
632        if let Some(call_id) = params
633            .get("call_id")
634            .and_then(|v| v.as_str())
635        {
636            let call = self.calls.lock().unwrap().get(call_id).cloned();
637            if let Some(call) = call {
638                call.dispatch_event(&event);
639
640                if call.current_state() == constants::CALL_STATE_ENDED {
641                    self.calls.lock().unwrap().remove(call_id);
642                }
643                return;
644            }
645        }
646
647        // Fire generic event handler if nothing else matched.
648        if let Some(handler) = self.on_event_handler.lock().unwrap().as_ref() {
649            handler(&event, outer_params);
650        }
651    }

738    /// Send a JSON-RPC request for a calling/messaging method.
739    ///
740    /// Mirrors Python's `RelayClient._send_request`: emits a flat-Blade
741    /// frame `{"method": <method>, "params": <params>}` directly — no
742    /// `signalwire.execute` wrapper. Both forms are accepted by the
743    /// production RELAY server and the mock; the flat form is what
744    /// every existing SDK port emits because it keeps the journal
745    /// filterable by inner method name without unwrapping.
746    ///
747    /// Returns the response's `result` value on success, or an `Err`
748    /// with the server's error message on failure or timeout.
749    pub fn execute_blocking(
750        &self,
751        method: &str,
752        inner_params: Value,
753    ) -> Result<Value, String> {
754        let id = generate_uuid();
755        let frame = json!({
756            "jsonrpc": "2.0",
757            "id": id,
758            "method": method,
759            "params": inner_params,
760        });
761
762        let (resp_tx, resp_rx) = mpsc::channel::<Result<Value, Value>>();
763        let resolve_tx = resp_tx.clone();
764        let reject_tx = resp_tx;
765        self.register_pending(
766            &id,
767            move |v| {
768                let _ = resolve_tx.send(Ok(v));
769            },
770            move |e| {
771                let _ = reject_tx.send(Err(e));
772            },
773        );
774
775        self.send(&frame);
776
777        // Use the same handshake-style timeout — 10s, matching Python's
778        // `_EXECUTE_TIMEOUT`.
779        match resp_rx.recv_timeout(HANDSHAKE_TIMEOUT) {
780            Ok(Ok(result)) => Ok(result),
781            Ok(Err(err)) => Err(err
782                .get("message")
783                .and_then(|v| v.as_str())
784                .unwrap_or("execute failed")
785                .to_string()),
786            Err(_) => {
787                self.pending.lock().unwrap().remove(&id);
788                Err(format!(
789                    "execute_blocking: timed out waiting for {} response",
790                    method
791                ))
792            }
793        }
794    }

796    /// Send an outbound SMS/MMS message.
797    ///
798    /// Mirrors Python's `RelayClient.send_message`. At least one of
799    /// `body` or `media` must be supplied. Returns a tracked `Message`
800    /// whose state will progress as `messaging.state` events arrive
801    /// from the server.
802    pub fn send_message_blocking(
803        &self,
804        to_number: &str,
805        from_number: &str,
806        body: Option<&str>,
807        media: Option<&[String]>,
808        tags: Option<&[String]>,
809        context: Option<&str>,
810    ) -> Result<Arc<Message>, String> {
811        if body.unwrap_or("").is_empty() && media.map(<[String]>::is_empty).unwrap_or(true) {
812            return Err("At least one of body or media is required".to_string());
813        }
814
815        let ctx = context
816            .map(str::to_string)
817            .or_else(|| self.protocol.lock().unwrap().clone())
818            .unwrap_or_else(|| "default".to_string());
819
820        let mut params = json!({
821            "context": ctx,
822            "to_number": to_number,
823            "from_number": from_number,
824        });
825        if let Some(b) = body {
826            if !b.is_empty() {
827                params["body"] = json!(b);
828            }
829        }
830        if let Some(m) = media {
831            if !m.is_empty() {
832                params["media"] = json!(m);
833            }
834        }
835        if let Some(t) = tags {
836            if !t.is_empty() {
837                params["tags"] = json!(t);
838            }
839        }
840
841        let result = self.execute_blocking("messaging.send", params)?;
842
843        let message_id = result
844            .get("message_id")
845            .and_then(|v| v.as_str())
846            .unwrap_or("")
847            .to_string();
848
849        // Build a Message we'll track.
850        let mut init_params = json!({
851            "context": ctx,
852            "direction": "outbound",
853            "from_number": from_number,
854            "to_number": to_number,
855            "state": "queued",
856        });
857        if !message_id.is_empty() {
858            init_params["message_id"] = json!(message_id);
859        }
860        if let Some(b) = body {
861            init_params["body"] = json!(b);
862        }
863        if let Some(m) = media {
864            init_params["media"] = json!(m);
865        }
866        if let Some(t) = tags {
867            init_params["tags"] = json!(t);
868        }
869
870        let msg = Arc::new(Message::new(&init_params));
871        if !message_id.is_empty() {
872            self.track_message(&message_id, msg.clone());
873        }
874        Ok(msg)
875    }

877    /// Initiate an outbound call using `calling.dial`.
878    ///
879    /// Mirrors Python's `RelayClient.dial`. The dial response carries no
880    /// `call_id` — the actual call info arrives via subsequent
881    /// `calling.call.dial` events keyed by `tag`. This method waits for
882    /// that event up to `dial_timeout` and returns the resolved Call.
883    ///
884    /// `devices` is the standard serial/parallel device matrix
885    /// (`[[device]]` for one parallel leg / serial = one inner list with
886    /// multiple devices).
887    pub fn dial_blocking(
888        self: &Arc<Self>,
889        devices: Value,
890        tag: Option<&str>,
891        max_duration: Option<u32>,
892        dial_timeout: Duration,
893    ) -> Result<Arc<Call>, String> {
894        let dial_tag = tag
895            .map(str::to_string)
896            .unwrap_or_else(generate_uuid);
897
898        let (resolve_tx, resolve_rx) = mpsc::channel::<Arc<Call>>();
899        // Fail channel — when the SDK observes a `failed` dial event for
900        // this tag we surface that here via handle_dial_event sending an
901        // Err. For now we wire only the success path; `failed` events
902        // are observed separately by reading the journal in tests.
903        self.register_dial(&dial_tag, move |call| {
904            let _ = resolve_tx.send(call);
905        });
906
907        // Track failure events: we install a generic event handler hook
908        // for the duration of this dial. The Client supports only a
909        // single `on_event` handler; saving and restoring the previous
910        // value lets nested dials work.
911        let dial_failed: Arc<Mutex<Option<String>>> = Arc::new(Mutex::new(None));
912        let _failed_rx = dial_failed.clone();
913
914        let mut params = json!({
915            "tag": dial_tag,
916            "devices": devices,
917        });
918        if let Some(md) = max_duration {
919            params["max_duration"] = json!(md);
920        }
921
922        // Send the dial RPC — response is just `{"code":"200","message":"Dialing"}`.
923        // We don't gate on its result; the actual completion comes via the
924        // dial event.
925        let _ = self.execute_blocking("calling.dial", params);
926
927        // Wait for either the resolved call or the timeout. The
928        // existing `handle_dial_event` resolves only on the
929        // `calling.call.dial` event with a `call.call_id` in `params.call`.
930        match resolve_rx.recv_timeout(dial_timeout) {
931            Ok(call) => Ok(call),
932            Err(_) => {
933                self.remove_pending_dial(&dial_tag);
934                if let Some(reason) = dial_failed.lock().unwrap().clone() {
935                    Err(format!("Dial failed: {}", reason))
936                } else {
937                    Err(format!(
938                        "Dial timed out waiting for answer (tag={})",
939                        dial_tag
940                    ))
941                }
942            }
943        }
944    }

950    fn handle_inbound_call(&self, event: &Event, params: &Value) {
951        let call_id = match params.get("call_id").and_then(|v| v.as_str()) {
952            Some(id) => id,
953            None => {
954                self.logger.warn("Inbound call event missing call_id");
955                return;
956            }
957        };
958
959        let call = Arc::new(Call::new(params));
960        self.calls
961            .lock()
962            .unwrap()
963            .insert(call_id.to_string(), call.clone());
964
965        self.logger
966            .info(&format!("Inbound call {}", call_id));
967
968        if let Some(handler) = self.on_call_handler.lock().unwrap().as_ref() {
969            handler(call, event);
970        }
971    }

973    fn handle_dial_event(&self, _event: &Event, params: &Value) {
974        let tag = match params.get("tag").and_then(|v| v.as_str()) {
975            Some(t) => t.to_string(),
976            None => return,
977        };
978
979        // The production server emits `params.call.call_id` (nested),
980        // while older shapes carried a top-level `params.call_id`. Try
981        // the nested shape first, then fall back. We also pass the
982        // nested `call` object as the construction params so the resulting
983        // Call carries device/tag fields too.
984        let call_obj = params.get("call");
985        let nested_call_id = call_obj.and_then(|c| c.get("call_id")).and_then(|v| v.as_str());
986        let top_level_call_id = params.get("call_id").and_then(|v| v.as_str());
987        let call_id = nested_call_id.or(top_level_call_id);
988
989        // Build the construction params: prefer the nested `call` object
990        // if it exists (it carries `tag`, `device`, etc. for the winner),
991        // otherwise fall back to the top-level params.
992        let ctor_params = call_obj.cloned().unwrap_or_else(|| params.clone());
993
994        // If the dial failed, surface a failed event but do not resolve
995        // the pending dial — the dial_blocking timeout path notes the
996        // failure. Same for any state other than answered: we wait for
997        // a winner.
998        let dial_state = params
999            .get("dial_state")
1000            .and_then(|v| v.as_str())
1001            .unwrap_or("");
1002
1003        if dial_state == "failed" {
1004            self.logger.warn(&format!("dial failed for tag={}", tag));
1005            self.pending_dials.lock().unwrap().remove(&tag);
1006            return;
1007        }
1008
1009        let call = if let Some(cid) = call_id {
1010            let mut calls = self.calls.lock().unwrap();
1011            if let Some(existing) = calls.get(cid) {
1012                existing.clone()
1013            } else {
1014                let call = Arc::new(Call::new(&ctor_params));
1015                calls.insert(cid.to_string(), call.clone());
1016                call
1017            }
1018        } else {
1019            return;
1020        };
1021
1022        // Resolve the pending dial
1023        let pending = self.pending_dials.lock().unwrap().remove(&tag);
1024        if let Some(dial) = pending {
1025            *call.dial_winner.lock().unwrap() = true;
1026            *call.state.lock().unwrap() = if dial_state.is_empty() {
1027                "answered".to_string()
1028            } else {
1029                dial_state.to_string()
1030            };
1031            (dial.resolve)(call);
1032        }
1033    }

1040    /// Long-running loop that owns the WebSocket. Alternates between
1041    /// short-timeout reads and draining the outbound write channel,
1042    /// dispatching every inbound text frame through `handle_message`.
1043    /// Exits when the closing flag is set, the write channel is
1044    /// dropped, or the socket reports a fatal error.
1045    fn reader_loop(
1046        client: Arc<Self>,
1047        mut socket: WsStream,
1048        write_rx: Receiver<WsMessage>,
1049    ) {
1050        // Track whether we observed a clean close so the disconnect
1051        // logic can decide whether to attempt a reconnect later.
1052        loop {
1053            if client.closing.load(Ordering::SeqCst) {
1054                let _ = socket.close(None);
1055                let _ = socket.flush();
1056                break;
1057            }
1058
1059            // 1) Drain any queued outbound frames before reading. Writes
1060            //    are infrequent; a single batch here keeps latency low
1061            //    and avoids stale frames when the server pushes a burst
1062            //    of events.
1063            let mut wrote_any = false;
1064            loop {
1065                match write_rx.try_recv() {
1066                    Ok(frame) => {
1067                        if let Err(e) = socket.send(frame) {
1068                            client
1069                                .logger
1070                                .warn(&format!("WS send error: {}", e));
1071                            break;
1072                        }
1073                        wrote_any = true;
1074                    }
1075                    Err(mpsc::TryRecvError::Empty) => break,
1076                    Err(mpsc::TryRecvError::Disconnected) => {
1077                        // Sender side dropped — disconnect() was called.
1078                        let _ = socket.close(None);
1079                        let _ = socket.flush();
1080                        return;
1081                    }
1082                }
1083            }
1084            if wrote_any {
1085                let _ = socket.flush();
1086            }
1087
1088            // 2) Try to read one inbound frame. The TCP read timeout is
1089            //    100ms so this returns regularly to give the write side
1090            //    a turn.
1091            match socket.read() {
1092                Ok(WsMessage::Text(t)) => {
1093                    client.handle_message(&t);
1094                }
1095                Ok(WsMessage::Binary(b)) => {
1096                    // RELAY uses text frames; binary is unexpected. Try
1097                    // to decode as UTF-8 and dispatch anyway, otherwise
1098                    // log and drop.
1099                    match std::str::from_utf8(&b) {
1100                        Ok(s) => client.handle_message(s),
1101                        Err(_) => client
1102                            .logger
1103                            .debug("ignoring non-UTF8 binary WS frame"),
1104                    }
1105                }
1106                Ok(WsMessage::Ping(p)) => {
1107                    // Tungstenite normally auto-pongs on the next write;
1108                    // ensure we explicitly respond so the server's
1109                    // half-open detector doesn't fire.
1110                    let _ = socket.send(WsMessage::Pong(p));
1111                }
1112                Ok(WsMessage::Pong(_)) => {
1113                    // No-op; we don't track our own pings here.
1114                }
1115                Ok(WsMessage::Close(_)) => {
1116                    client.logger.info("server closed WS");
1117                    *client.connected.lock().unwrap() = false;
1118                    break;
1119                }
1120                Ok(WsMessage::Frame(_)) => {
1121                    // Raw frame — should not happen with the protocol
1122                    // module's `read()` API. Ignore.
1123                }
1124                Err(tungstenite::Error::Io(e))
1125                    if e.kind() == std::io::ErrorKind::WouldBlock
1126                        || e.kind() == std::io::ErrorKind::TimedOut =>
1127                {
1128                    // Read timeout — loop back to drain writes. This is
1129                    // the steady state when the server is quiet.
1130                }
1131                Err(tungstenite::Error::ConnectionClosed) => {
1132                    client.logger.info("WS connection closed");
1133                    *client.connected.lock().unwrap() = false;
1134                    break;
1135                }
1136                Err(tungstenite::Error::AlreadyClosed) => {
1137                    *client.connected.lock().unwrap() = false;
1138                    break;
1139                }
1140                Err(e) => {
1141                    client.logger.warn(&format!("WS read error: {}", e));
1142                    *client.connected.lock().unwrap() = false;
1143                    break;
1144                }
1145            }
1146        }
1147
1148        // Reader is exiting — clear the write sender so subsequent
1149        // `send()` calls fall back to in-memory buffering rather than
1150        // pushing into a dead channel.
1151        *client.write_tx.lock().unwrap() = None;
1152    }
1153}

1155/// Generate a simple UUID v4.
1156fn generate_uuid() -> String {
1157    use rand::Rng;
1158    let mut rng = rand::thread_rng();
1159    let mut data = [0u8; 16];
1160    rng.fill(&mut data);
1161    data[6] = (data[6] & 0x0f) | 0x40;
1162    data[8] = (data[8] & 0x3f) | 0x80;
1163    format!(
1164        "{:08x}-{:04x}-{:04x}-{:04x}-{:012x}",
1165        u32::from_be_bytes([data[0], data[1], data[2], data[3]]),
1166        u16::from_be_bytes([data[4], data[5]]),
1167        u16::from_be_bytes([data[6], data[7]]),
1168        u16::from_be_bytes([data[8], data[9]]),
1169        ((data[10] as u64) << 40)
1170            | ((data[11] as u64) << 32)
1171            | ((data[12] as u64) << 24)
1172            | ((data[13] as u64) << 16)
1173            | ((data[14] as u64) << 8)
1174            | (data[15] as u64),
1175    )
1176}

1242        // Server thread: accept one upgrade, run the handshake, push a
1243        // response, and exit.
1244        let server = std::thread::spawn(move || {
1245            let (stream, _) = listener.accept().unwrap();
1246            let mut ws = tungstenite::accept(stream).unwrap();
1247            // Read the connect frame.
1248            loop {
1249                match ws.read() {
1250                    Ok(WsMessage::Text(t)) => {
1251                        let v: Value = serde_json::from_str(&t).unwrap();
1252                        if v.get("method").and_then(|m| m.as_str()) == Some("signalwire.connect") {
1253                            let id = v.get("id").cloned().unwrap_or(json!(""));
1254                            let project = v
1255                                .get("params")
1256                                .and_then(|p| p.get("authentication"))
1257                                .and_then(|a| a.get("project"))
1258                                .cloned()
1259                                .unwrap_or(json!(""));
1260                            let resp = json!({
1261                                "jsonrpc": "2.0",
1262                                "id": id,
1263                                "result": {
1264                                    "authorization": {
1265                                        "authorization_state": "fixture-state-token",
1266                                    },
1267                                    "protocol": "signalwire-relay-test",
1268                                    "project": project,
1269                                }
1270                            });
1271                            ws.send(WsMessage::Text(resp.to_string())).unwrap();
1272                            // Hold the connection open briefly so the
1273                            // client doesn't see a premature close.
1274                            std::thread::sleep(Duration::from_millis(150));
1275                            break;
1276                        }
1277                    }
1278                    Ok(_) => {}
1279                    Err(_) => break,
1280                }
1281            }
1282        });
1283
1284        // SAFETY: this test serializes itself through env-var locking
1285        // by being one of the only tests that mutates the relay env
1286        // vars. Other tests in this module don't touch them.
1287        unsafe {
1288            std::env::set_var("SIGNALWIRE_RELAY_SCHEME", "ws");
1289            std::env::set_var("SIGNALWIRE_RELAY_HOST", format!("127.0.0.1:{}", port));
1290        }
1291
1292        let client = Arc::new(Client::new("test-project", "test-token", "ignored"));
1293        let res = client.connect();
1294        assert!(res.is_ok(), "connect failed: {:?}", res);
1295        // Authorization state should have been captured from the
1296        // fixture's response — proves the client parsed
1297        // `result.authorization.authorization_state`.
1298        assert_eq!(
1299            *client.authorization_state.lock().unwrap(),
1300            Some("fixture-state-token".to_string())
1301        );
1302        // Protocol string captured.
1303        assert_eq!(
1304            *client.protocol.lock().unwrap(),
1305            Some("signalwire-relay-test".to_string())
1306        );
1307        client.disconnect();
1308        let _ = server.join();
1309
1310        unsafe {
1311            std::env::remove_var("SIGNALWIRE_RELAY_SCHEME");
1312            std::env::remove_var("SIGNALWIRE_RELAY_HOST");
1313        }
1314    }
