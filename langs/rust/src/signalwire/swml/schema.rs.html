3use serde_json::Value;

5/// Metadata about a single SWML verb extracted from the schema.
6#[derive(Debug, Clone)]
7pub struct VerbInfo {
8    pub name: String,
9    pub schema_name: String,
10}
11
12struct SchemaData {
13    verbs: Vec<VerbInfo>,
14}
15
16static SCHEMA: LazyLock<SchemaData> = LazyLock::new(|| {
17    let raw = include_str!("schema.json");
18    let data: Value = serde_json::from_str(raw).expect("schema.json must be valid JSON");
19
20    let defs = match data.get("$defs") {
21        Some(d) => d,
22        None => return SchemaData { verbs: Vec::new() },
23    };
24
25    let any_of = match defs.get("SWMLMethod").and_then(|m| m.get("anyOf")) {
26        Some(Value::Array(arr)) => arr,
27        _ => return SchemaData { verbs: Vec::new() },
28    };
29
30    let mut verbs = Vec::new();
31
32    for entry in any_of {
33        let ref_str = match entry.get("$ref").and_then(|r| r.as_str()) {
34            Some(s) => s,
35            None => continue,
36        };
37
38        // e.g. "#/$defs/Answer" -> "Answer"
39        let def_name = match ref_str.rsplit('/').next() {
40            Some(n) => n,
41            None => continue,
42        };
43
44        let defn = match defs.get(def_name) {
45            Some(d) => d,
46            None => continue,
47        };
48
49        let props = match defn.get("properties").and_then(|p| p.as_object()) {
50            Some(p) if !p.is_empty() => p,
51            _ => continue,
52        };
53
54        // The first property key is the actual verb name
55        let actual_verb = match props.keys().next() {
56            Some(k) => k.clone(),
57            None => continue,
58        };
59
60        verbs.push(VerbInfo {
61            name: actual_verb,
62            schema_name: def_name.to_string(),
63        });
64    }
65
66    SchemaData { verbs }
67});
68
69/// Check whether a verb name is valid.
70pub fn is_valid_verb(name: &str) -> bool {
71    SCHEMA.verbs.iter().any(|v| v.name == name)
72}
73
74/// Get sorted list of all verb names.
75pub fn get_verb_names() -> Vec<String> {
76    let mut names: Vec<String> = SCHEMA.verbs.iter().map(|v| v.name.clone()).collect();
77    names.sort();
78    names
79}
80
81/// Get verb metadata, or `None` if not found.
82pub fn get_verb(name: &str) -> Option<&VerbInfo> {
83    SCHEMA.verbs.iter().find(|v| v.name == name)
84}
85
86/// Number of verbs defined in the schema.
87pub fn verb_count() -> usize {
88    SCHEMA.verbs.len()
89}
