42    fn register_tools(&self, agent: &mut AgentBase) {
43        let prefix = self.sp.get_str_or("prefix", "");
44        let questions = self.sp.get_array("questions");
45        let completion_message = self.sp.get_str_or(
46            "completion_message",
47            "All questions have been answered. Thank you!",
48        );
49        let namespace = self.get_instance_key();
50
51        let start_tool = if prefix.is_empty() {
52            "start_questions".to_string()
53        } else {
54            format!("{prefix}_start_questions")
55        };
56        let submit_tool = if prefix.is_empty() {
57            "submit_answer".to_string()
58        } else {
59            format!("{prefix}_submit_answer")
60        };
61
62        // start_questions tool
63        let questions_clone = questions.clone();
64        let namespace_clone = namespace.clone();
65        agent.define_tool(
66            &start_tool,
67            "Start the question gathering process and get the first question",
68            json!({}),
69            Box::new(move |_args, _raw| {
70                let mut result = FunctionResult::new();
71
72                if questions_clone.is_empty() {
73                    result.set_response("No questions configured.");
74                    return result;
75                }
76
77                let first = questions_clone[0]
78                    .get("question_text")
79                    .and_then(|v| v.as_str())
80                    .unwrap_or("No question text.");
81
82                result.set_response(&format!("Starting questions. First question: {first}"));
83                result.add_action(json!({
84                    "set_global_data": {
85                        namespace_clone.clone(): {
86                            "questions": questions_clone,
87                            "question_index": 0,
88                            "answers": [],
89                        }
90                    }
91                }));
92
93                result
94            }),
95            false,
96        );
97
98        // submit_answer tool
99        let questions_clone2 = questions.clone();
100        let completion_msg = completion_message.clone();
101        agent.define_tool(
102            &submit_tool,
103            "Submit an answer to the current question",
104            json!({
105                "answer": {
106                    "type": "string",
107                    "description": "The answer to the current question",
108                    "required": true,
109                },
110                "confirmed_by_user": {
111                    "type": "boolean",
112                    "description": "Whether the user has confirmed this answer is correct",
113                },
114            }),
115            Box::new(move |args, _raw| {
116                let mut result = FunctionResult::new();
117                let answer = args
118                    .get("answer")
119                    .and_then(|v| v.as_str())
120                    .unwrap_or("");
121                let confirmed = args
122                    .get("confirmed_by_user")
123                    .and_then(serde_json::Value::as_bool)
124                    .unwrap_or(false);
125
126                if answer.is_empty() {
127                    result.set_response("Please provide an answer.");
128                    return result;
129                }
130
131                let total = questions_clone2.len();
132                let current_index = 0_usize; // Managed by global data in runtime
133
134                let current_question = questions_clone2.get(current_index);
135                let needs_confirm = current_question
136                    .and_then(|q| q.get("confirm"))
137                    .and_then(serde_json::Value::as_bool)
138                    .unwrap_or(false);
139
140                if needs_confirm && !confirmed {
141                    let question_text = current_question
142                        .and_then(|q| q.get("question_text"))
143                        .and_then(|v| v.as_str())
144                        .unwrap_or("");
145                    result.set_response(&format!(
146                        "You answered \"{answer}\" for: {question_text}. Can you confirm this is correct?"
147                    ));
148                    return result;
149                }
150
151                let next_index = current_index + 1;
152                if next_index >= total {
153                    result.set_response(&completion_msg);
154                } else {
155                    let next_question = questions_clone2[next_index]
156                        .get("question_text")
157                        .and_then(|v| v.as_str())
158                        .unwrap_or("No question text.");
159                    result.set_response(&format!(
160                        "Answer recorded. Next question: {next_question}"
161                    ));
162                }
163
164                result
165            }),
166            false,
167        );
168    }

170    fn get_global_data(&self) -> Map<String, Value> {
171        let namespace = self.get_instance_key();
172        let questions = self.sp.get_array("questions");
173
174        let mut data = Map::new();
175        data.insert(
176            namespace,
177            json!({
178                "questions": questions,
179                "question_index": 0,
180                "answers": [],
181            }),
182        );
183        data
184    }
185
186    fn get_prompt_sections(&self) -> Vec<Value> {
187        if self.sp.get_bool("skip_prompt") {
188            return Vec::new();
189        }
190
191        let instance_key = self.get_instance_key();
192        let questions = self.sp.get_array("questions");
193
194        let mut bullets = vec![
195            "Call start_questions to begin the question flow.".to_string(),
196            "Submit each answer using submit_answer with the user's response.".to_string(),
197            "Questions that require confirmation will ask the user to verify their answer."
198                .to_string(),
199        ];
200
201        for q in &questions {
202            if let Some(prompt_add) = q.get("prompt_add").and_then(|v| v.as_str())
203                && !prompt_add.is_empty()
204            {
205                bullets.push(prompt_add.to_string());
206            }
207        }
208
209        let bullet_values: Vec<Value> = bullets.into_iter().map(|b| json!(b)).collect();
210
211        vec![json!({
212            "title": format!("Info Gatherer ({})", instance_key),
213            "body": "You need to gather information from the user by asking a series of questions.",
214            "bullets": bullet_values,
215        })]
216    }
