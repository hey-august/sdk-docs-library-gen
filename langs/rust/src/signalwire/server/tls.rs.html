8//! Shared HTTPS binding for the SDK's webhook / SWML / agent HTTP servers.
9//!
10//! Python's `SecurityConfig` reads `SWML_SSL_ENABLED` / `SWML_SSL_CERT_PATH` /
11//! `SWML_SSL_KEY_PATH` (see `core/security_config.py`) and, when enabled, hands
12//! `ssl_certfile` / `ssl_keyfile` to uvicorn so the server speaks real HTTPS.
13//! The Rust port serves with `tiny_http`; this helper mirrors the *same env
14//! contract* and binds either a plain (`Server::http`) or a TLS
15//! (`Server::https`, via `tiny_http`'s `ssl-rustls` feature) listener.
16//!
17//! Kept as a `pub(crate)` helper so the three server entry points
18//! (`AgentServer::run`, `Service::run`, `AgentBase::run`) share one code path
19//! and the public surface is unchanged — TLS is configured entirely through
20//! the documented `SWML_SSL_*` environment variables, no new public method.
21
22use std::env;
23
24use super::error::ServerError;
25
26/// Env vars mirroring Python `SecurityConfig` (`core/security_config.py`).
27const SSL_ENABLED_ENV: &str = "SWML_SSL_ENABLED";
28const SSL_CERT_PATH_ENV: &str = "SWML_SSL_CERT_PATH";
29const SSL_KEY_PATH_ENV: &str = "SWML_SSL_KEY_PATH";
30
31/// Resolved TLS material: PEM cert + key file *contents*.
32struct TlsMaterial {
33    certificate: Vec<u8>,
34    private_key: Vec<u8>,
35}
36
37/// Read `SWML_SSL_*`. Returns `Ok(Some(..))` only when SSL is enabled *and*
38/// both cert and key paths are set and readable. Returns `Ok(None)` when SSL
39/// is off (the normal HTTP path). Returns `Err` when SSL is requested but the
40/// configuration is incomplete / unreadable — surfaced to the caller so a
41/// misconfigured deployment fails loudly instead of silently serving HTTP.
42fn resolve_tls_material() -> Result<Option<TlsMaterial>, ServerError> {
43    let enabled = env::var(SSL_ENABLED_ENV)
44        .is_ok_and(|v| matches!(v.to_ascii_lowercase().as_str(), "true" | "1" | "yes"));
45    if !enabled {
46        return Ok(None);
47    }
48
49    let cert_path = env::var(SSL_CERT_PATH_ENV).ok().filter(|s| !s.is_empty());
50    let key_path = env::var(SSL_KEY_PATH_ENV).ok().filter(|s| !s.is_empty());
51    let (Some(cert_path), Some(key_path)) = (cert_path, key_path) else {
52        return Err(ServerError::TlsConfig {
53            message: format!(
54                "{SSL_ENABLED_ENV} is set but {SSL_CERT_PATH_ENV} / {SSL_KEY_PATH_ENV} are not both provided"
55            ),
56        });
57    };
58
59    let certificate = std::fs::read(&cert_path).map_err(|e| ServerError::TlsConfig {
60        message: format!("read {SSL_CERT_PATH_ENV} {cert_path}: {e}"),
61    })?;
62    let private_key = std::fs::read(&key_path).map_err(|e| ServerError::TlsConfig {
63        message: format!("read {SSL_KEY_PATH_ENV} {key_path}: {e}"),
64    })?;
65    Ok(Some(TlsMaterial {
66        certificate,
67        private_key,
68    }))
69}
70
71/// Bind the server at `addr`, choosing HTTPS when `SWML_SSL_*` requests it and
72/// plain HTTP otherwise. Returns `(server, is_https)` so the caller can log the
73/// correct scheme.
74///
75/// Panics (consistent with the existing `run()` bind-failure behavior) on a
76/// genuine bind error; SSL-misconfiguration is reported through the returned
77/// `Result` so callers can choose how to fail.
78pub(crate) fn bind_server(addr: &str) -> Result<(tiny_http::Server, bool), ServerError> {
79    if let Some(material) = resolve_tls_material()? {
80        let config = tiny_http::SslConfig {
81            certificate: material.certificate,
82            private_key: material.private_key,
83        };
84        let server = tiny_http::Server::https(addr, config).map_err(|e| ServerError::Bind {
85            addr: addr.to_string(),
86            source: e.to_string(),
87        })?;
88        Ok((server, true))
89    } else {
90        let server = tiny_http::Server::http(addr).map_err(|e| ServerError::Bind {
91            addr: addr.to_string(),
92            source: e.to_string(),
93        })?;
94        Ok((server, false))
95    }
96}
