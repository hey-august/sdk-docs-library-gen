14/// CLI entry point for the `swaig-test` tool.
15///
16/// Usage:
17///   swaig-test --example <NAME> --list-tools
18///   swaig-test --url <URL> [options]
19///
20/// Options:
21///   --example <NAME>     `SWMLService` example to introspect by name (e.g.
22///                        `swmlservice_swaig_standalone`). Runs the example
23///                        in-process via `cargo run --example` with the
24///                        `SWAIG_LIST_TOOLS=1` env var; the SDK's `serve()`
25///                        dumps the runtime registry instead of binding a
26///                        port.
27///   --url <URL>          SWAIG endpoint URL. Basic auth can be embedded as
28///                        user:pass@host.
29///   --dump-swml          Fetch and dump the SWML document (URL mode).
30///   --list-tools         List available SWAIG tools.
31///   --exec <TOOL>        Execute a specific SWAIG tool by name (URL mode).
32///   --param <K=V>        Parameter for --exec (repeatable).
33///   --raw                Print raw JSON responses (no formatting).
34///   --verbose            Enable verbose output.
35///   --help               Print this help message.
36fn main() {
37    let args: Vec<String> = env::args().collect();
38
39    if args.len() < 2 || args.contains(&"--help".to_string()) {
40        print_help();
41        process::exit(0);
42    }
43
44    let mut url: Option<String> = None;
45    let mut example: Option<String> = None;
46    let mut dump_swml = false;
47    let mut list_tools = false;
48    let mut exec_tool: Option<String> = None;
49    let mut params: Vec<(String, String)> = Vec::new();
50    let mut raw = false;
51    let mut verbose = false;
52
53    let mut i = 1;
54    while i < args.len() {
55        match args[i].as_str() {
56            "--url" => {
57                i += 1;
58                if i < args.len() {
59                    url = Some(args[i].clone());
60                } else {
61                    eprintln!("Error: --url requires a value");
62                    process::exit(1);
63                }
64            }
65            "--example" => {
66                i += 1;
67                if i < args.len() {
68                    example = Some(args[i].clone());
69                } else {
70                    eprintln!("Error: --example requires a name");
71                    process::exit(1);
72                }
73            }
74            "--dump-swml" => dump_swml = true,
75            "--list-tools" => list_tools = true,
76            "--exec" => {
77                i += 1;
78                if i < args.len() {
79                    exec_tool = Some(args[i].clone());
80                } else {
81                    eprintln!("Error: --exec requires a tool name");
82                    process::exit(1);
83                }
84            }
85            "--param" => {
86                i += 1;
87                if i < args.len() {
88                    if let Some(eq_pos) = args[i].find('=') {
89                        let k = args[i][..eq_pos].to_string();
90                        let v = args[i][eq_pos + 1..].to_string();
91                        params.push((k, v));
92                    } else {
93                        eprintln!("Error: --param requires K=V format");
94                        process::exit(1);
95                    }
96                } else {
97                    eprintln!("Error: --param requires a value");
98                    process::exit(1);
99                }
100            }
101            "--raw" => raw = true,
102            "--verbose" => verbose = true,
103            "--help" => {
104                print_help();
105                process::exit(0);
106            }
107            other => {
108                eprintln!("Error: unknown option: {other}");
109                process::exit(1);
110            }
111        }
112        i += 1;
113    }
114
115    // Example/file-loader mode.
116    if let Some(name) = example {
117        if !list_tools {
118            eprintln!("Error: --example currently only supports --list-tools");
119            process::exit(1);
120        }
121        do_list_tools_via_introspect(&name, raw, verbose);
122        return;
123    }
124
125    let Some(url) = url else {
126        eprintln!("Error: --url or --example is required");
127        process::exit(1);
128    };
129
130    // Extract auth from URL if embedded
131    let (base_url, auth_header) = extract_url_auth(&url);
132
133    if verbose {
134        eprintln!("[verbose] URL: {base_url}");
135        if auth_header.is_some() {
136            eprintln!("[verbose] Auth: (embedded credentials)");
137        }
138    }
139
140    // Route to the appropriate action
141    if dump_swml {
142        do_dump_swml(&base_url, auth_header.as_deref(), raw, verbose);
143    } else if list_tools {
144        do_list_tools(&base_url, auth_header.as_deref(), raw, verbose);
145    } else if let Some(tool) = exec_tool {
146        do_exec_tool(
147            &base_url,
148            auth_header.as_deref(),
149            &tool,
150            &params,
151            raw,
152            verbose,
153        );
154    } else {
155        eprintln!("Error: specify --dump-swml, --list-tools, or --exec <tool>");
156        process::exit(1);
157    }
158}

186/// Introspect a `SWMLService` example by spawning `cargo run --example NAME`
187/// with `SWAIG_LIST_TOOLS=1`. The SDK's `Service::run()` honors that env var
188/// by printing the registry to stdout between sentinels and exiting before
189/// it would have bound any port. We capture stdout, slice out the JSON, and
190/// pretty-print or emit raw.
191fn do_list_tools_via_introspect(example_name: &str, raw: bool, verbose: bool) {
192    if verbose {
193        eprintln!("[verbose] running `cargo run --example {example_name}` with SWAIG_LIST_TOOLS=1");
194    }
195    let mut cmd = std::process::Command::new("cargo");
196    cmd.args(["run", "--quiet", "--example", example_name])
197        .env("SWAIG_LIST_TOOLS", "1");
198    let output = match cmd.output() {
199        Ok(o) => o,
200        Err(e) => {
201            eprintln!("Error: failed to spawn cargo: {e}");
202            process::exit(1);
203        }
204    };
205    if !output.status.success() {
206        let stderr = String::from_utf8_lossy(&output.stderr);
207        eprintln!("Error: example `{example_name}` exited non-zero");
208        if !stderr.is_empty() {
209            eprintln!("--- cargo stderr ---\n{}", stderr.trim_end());
210        }
211        process::exit(1);
212    }
213    let stdout = String::from_utf8_lossy(&output.stdout);
214    let Some(body) = extract_introspect_payload(&stdout) else {
215        eprintln!(
216            "Error: example `{example_name}` did not emit __SWAIG_TOOLS_BEGIN__/__SWAIG_TOOLS_END__ markers. Make sure it calls service.run()."
217        );
218        if verbose {
219            eprintln!("--- raw stdout ---\n{stdout}");
220        }
221        process::exit(1);
222    };
223    let parsed: Value = match serde_json::from_str(body) {
224        Ok(v) => v,
225        Err(e) => {
226            eprintln!("Error: malformed introspect payload: {e}");
227            eprintln!("--- payload ---\n{body}");
228            process::exit(1);
229        }
230    };
231    if raw {
232        println!("{body}");
233        return;
234    }
235    let tools = parsed.get("tools").and_then(|v| v.as_array());
236    let Some(tools) = tools else {
237        println!(
238            "{}",
239            serde_json::to_string_pretty(&parsed).unwrap_or_default()
240        );
241        return;
242    };
243    if tools.is_empty() {
244        println!("No tools registered.");
245        return;
246    }
247    println!("Registered SWAIG tools ({}):", tools.len());
248    for (i, tool) in tools.iter().enumerate() {
249        let name = tool
250            .get("function")
251            .and_then(|v| v.as_str())
252            .or_else(|| tool.get("name").and_then(|v| v.as_str()))
253            .unwrap_or("<unnamed>");
254        let desc = tool
255            .get("purpose")
256            .and_then(|v| v.as_str())
257            .or_else(|| tool.get("description").and_then(|v| v.as_str()))
258            .unwrap_or("");
259        println!("  {}. {} — {}", i + 1, name, desc);
260        let argument = tool.get("argument").or_else(|| tool.get("parameters"));
261        if let Some(arg) = argument
262            && let Some(props) = arg.get("properties").and_then(|v| v.as_object())
263        {
264            for (pname, pdef) in props {
265                let ptype = pdef.get("type").and_then(|v| v.as_str()).unwrap_or("");
266                let pdesc = pdef
267                    .get("description")
268                    .and_then(|v| v.as_str())
269                    .unwrap_or("");
270                println!("       - {pname} ({ptype}): {pdesc}");
271            }
272        }
273    }
274}

276/// Extract the JSON payload between `__SWAIG_TOOLS_BEGIN__` and
277/// `__SWAIG_TOOLS_END__` markers in the example's stdout. Returns None if
278/// either marker is missing.
279fn extract_introspect_payload(stdout: &str) -> Option<&str> {
280    let begin = stdout.find("__SWAIG_TOOLS_BEGIN__")?;
281    let after_begin = &stdout[begin + "__SWAIG_TOOLS_BEGIN__".len()..];
282    let end = after_begin.find("__SWAIG_TOOLS_END__")?;
283    Some(after_begin[..end].trim())
284}

286/// Extract Basic auth credentials from a URL of the form
287/// `http://user:pass@host:port/path` and return the cleaned URL + auth header.
288fn extract_url_auth(url: &str) -> (String, Option<String>) {
289    // Find the scheme separator
290    let scheme_end = match url.find("://") {
291        Some(pos) => pos + 3,
292        None => return (url.to_string(), None),
293    };
294
295    let scheme = &url[..scheme_end];
296    let rest = &url[scheme_end..];
297
298    // Look for @ in the authority section (before path)
299    let authority_end = rest.find('/').unwrap_or(rest.len());
300    let authority = &rest[..authority_end];
301    let path_and_rest = &rest[authority_end..];
302
303    if let Some(at_pos) = authority.find('@') {
304        let user_pass = &authority[..at_pos];
305        let host = &authority[at_pos + 1..];
306
307        let auth = format!("Basic {}", BASE64.encode(user_pass));
308        let clean_url = format!("{scheme}{host}{path_and_rest}");
309
310        (clean_url, Some(auth))
311    } else {
312        (url.to_string(), None)
313    }
314}

316/// Build request headers with optional auth.
317fn build_headers(auth: Option<&str>) -> HashMap<String, String> {
318    let mut headers = HashMap::new();
319    headers.insert("Content-Type".to_string(), "application/json".to_string());
320    headers.insert("Accept".to_string(), "application/json".to_string());
321    if let Some(a) = auth {
322        headers.insert("Authorization".to_string(), a.to_string());
323    }
324    headers
325}

327/// Make an HTTP request via `ureq`.
328///
329/// 4xx / 5xx are not transport failures: the agent returns the status code
330/// and body verbatim and lets the caller decide how to react. The only
331/// `Err` cases are genuine I/O / DNS / TLS / parse errors that prevented
332/// us from getting a status line back.
333///
334/// 30-second global timeout. Verbose mode mirrors method/URL/headers/body
335/// to stderr before the call.
336fn http_request(
337    method: &str,
338    url: &str,
339    headers: &HashMap<String, String>,
340    body: Option<&str>,
341    verbose: bool,
342) -> Result<(u16, String), String> {
343    if verbose {
344        eprintln!("[verbose] {method} {url}");
345        for (k, v) in headers {
346            eprintln!("[verbose]   {k}: {v}");
347        }
348        if let Some(b) = body {
349            eprintln!("[verbose]   body: {b}");
350        }
351    }
352
353    let agent: ureq::Agent = ureq::Agent::config_builder()
354        .timeout_global(Some(std::time::Duration::from_secs(30)))
355        .http_status_as_error(false)
356        .build()
357        .into();
358
359    let response_result = match method.to_ascii_uppercase().as_str() {
360        "GET" => {
361            let mut req = agent.get(url);
362            for (k, v) in headers {
363                req = req.header(k, v);
364            }
365            req.call()
366        }
367        "POST" => {
368            let mut req = agent.post(url);
369            for (k, v) in headers {
370                req = req.header(k, v);
371            }
372            match body {
373                Some(b) => req.send(b),
374                None => req.send_empty(),
375            }
376        }
377        "PUT" => {
378            let mut req = agent.put(url);
379            for (k, v) in headers {
380                req = req.header(k, v);
381            }
382            match body {
383                Some(b) => req.send(b),
384                None => req.send_empty(),
385            }
386        }
387        other => {
388            return Err(format!("Unsupported HTTP method: {other}"));
389        }
390    };
391
392    let mut response = response_result.map_err(|e| format!("HTTP {method} {url} failed: {e}"))?;
393    let status = response.status().as_u16();
394    let body_str = response
395        .body_mut()
396        .read_to_string()
397        .map_err(|e| format!("HTTP {method} {url} body read failed: {e}"))?;
398    Ok((status, body_str))
399}

401fn do_dump_swml(base_url: &str, auth: Option<&str>, raw: bool, verbose: bool) {
402    let headers = build_headers(auth);
403    match http_request("GET", base_url, &headers, None, verbose) {
404        Ok((_status, body)) => {
405            if raw {
406                println!("{body}");
407            } else {
408                match serde_json::from_str::<Value>(&body) {
409                    Ok(v) => println!("{}", serde_json::to_string_pretty(&v).unwrap()),
410                    Err(_) => println!("{body}"),
411                }
412            }
413        }
414        Err(e) => {
415            eprintln!("Error: {e}");
416            process::exit(1);
417        }
418    }
419}

421fn do_list_tools(base_url: &str, auth: Option<&str>, raw: bool, verbose: bool) {
422    let swaig_url = format!("{}/swaig", base_url.trim_end_matches('/'));
423    let headers = build_headers(auth);
424    match http_request("GET", &swaig_url, &headers, None, verbose) {
425        Ok((_status, body)) => {
426            if raw {
427                println!("{body}");
428            } else {
429                match serde_json::from_str::<Value>(&body) {
430                    Ok(v) => {
431                        if let Some(arr) = v.as_array() {
432                            if arr.is_empty() {
433                                println!("No tools available.");
434                            } else {
435                                for (i, tool) in arr.iter().enumerate() {
436                                    let name = tool
437                                        .get("function")
438                                        .and_then(|f| f.get("name"))
439                                        .and_then(|n| n.as_str())
440                                        .or_else(|| tool.get("name").and_then(|n| n.as_str()))
441                                        .unwrap_or("<unnamed>");
442                                    let desc = tool
443                                        .get("function")
444                                        .and_then(|f| f.get("description"))
445                                        .and_then(|d| d.as_str())
446                                        .or_else(|| {
447                                            tool.get("description").and_then(|d| d.as_str())
448                                        })
449                                        .unwrap_or("");
450                                    println!("  {}. {} - {}", i + 1, name, desc);
451                                }
452                            }
453                        } else {
454                            println!("{}", serde_json::to_string_pretty(&v).unwrap());
455                        }
456                    }
457                    Err(_) => println!("{body}"),
458                }
459            }
460        }
461        Err(e) => {
462            eprintln!("Error: {e}");
463            process::exit(1);
464        }
465    }
466}

468fn do_exec_tool(
469    base_url: &str,
470    auth: Option<&str>,
471    tool: &str,
472    params: &[(String, String)],
473    raw: bool,
474    verbose: bool,
475) {
476    let swaig_url = format!("{}/swaig", base_url.trim_end_matches('/'));
477    let headers = build_headers(auth);
478
479    // Build argument object
480    let mut args = serde_json::Map::new();
481    for (k, v) in params {
482        // Try to parse as JSON first, fall back to string
483        let val: Value = serde_json::from_str(v).unwrap_or(Value::String(v.clone()));
484        args.insert(k.clone(), val);
485    }
486
487    let body = json!({
488        "action": "execute",
489        "function": tool,
490        "argument": {
491            "parsed": [args],
492        },
493    });
494
495    let body_str = serde_json::to_string(&body).unwrap();
496
497    match http_request("POST", &swaig_url, &headers, Some(&body_str), verbose) {
498        Ok((_status, resp_body)) => {
499            if raw {
500                println!("{resp_body}");
501            } else {
502                match serde_json::from_str::<Value>(&resp_body) {
503                    Ok(v) => println!("{}", serde_json::to_string_pretty(&v).unwrap()),
504                    Err(_) => println!("{resp_body}"),
505                }
506            }
507        }
508        Err(e) => {
509            eprintln!("Error: {e}");
510            process::exit(1);
511        }
512    }
513}

592    /// Spawn a `tiny_http` server on an ephemeral port that responds with the
593    /// given fixed status + body, capturing whatever the request was. Returns
594    /// (`base_url`, `request_capture`). Used by the `http_request` behavior tests
595    /// below. Killed when the returned guard drops.
596    fn spawn_test_server(
597        status: u16,
598        response_body: &'static str,
599    ) -> (String, CapturedRequests, std::thread::JoinHandle<()>) {
600        let server = tiny_http::Server::http("127.0.0.1:0").expect("bind 127.0.0.1:0");
601        let port = server.server_addr().to_ip().unwrap().port();
602        let base = format!("http://127.0.0.1:{port}");
603        let captured = std::sync::Arc::new(std::sync::Mutex::new(Vec::new()));
604        let cap_clone = std::sync::Arc::clone(&captured);
605        let handle = std::thread::spawn(move || {
606            for mut req in server.incoming_requests() {
607                let method = req.method().as_str().to_string();
608                let path = req.url().to_string();
609                let mut hmap = HashMap::new();
610                for h in req.headers() {
611                    hmap.insert(
612                        h.field.as_str().as_str().to_string(),
613                        h.value.as_str().to_string(),
614                    );
615                }
616                let mut body = String::new();
617                let _ = req.as_reader().read_to_string(&mut body);
618                cap_clone.lock().unwrap().push((method, path, hmap, body));
619                let resp = tiny_http::Response::from_string(response_body).with_status_code(status);
620                let _ = req.respond(resp);
621            }
622        });
623        (base, captured, handle)
624    }

652        let (status, resp) = http_request("POST", &url, &headers, Some(body), false)

654        assert_eq!(status, 200);
655        assert!(resp.contains("\"response\":\"ACME\""));
656        let cap = captured.lock().unwrap();
657        assert_eq!(cap[0].0, "POST");
658        assert_eq!(cap[0].1, "/swaig");
659        // Auth header forwarded verbatim. Header names are case-insensitive
660        // in HTTP — different stacks normalize differently — so look it up
661        // case-insensitively.
662        let auth = cap[0]
663            .2
664            .iter()
665            .find(|(k, _)| k.eq_ignore_ascii_case("Authorization"))
666            .map(|(_, v)| v.as_str());
667        assert_eq!(auth, Some("Basic dGVzdDp0ZXN0"));
668        // Body forwarded verbatim.
669        assert!(cap[0].3.contains("ACME"));
670    }
