Overview

def self.from_payload(payload)
  et = payload['event_type'] || ''
  p  = payload['params'] || {}
  new(
    event_type: et,
    params:     p,
    call_id:    p['call_id'] || '',
    timestamp:  p['timestamp'] || 0.0
  )
end
