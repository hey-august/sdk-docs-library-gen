def to_h
  h = { "key" => @key, "question" => @question }
  h["type"]      = @type      if @type != 'string'
  h["confirm"]   = true       if @confirm
  h["prompt"]    = @prompt    if @prompt
  h["functions"] = @functions if @functions
  h
end
