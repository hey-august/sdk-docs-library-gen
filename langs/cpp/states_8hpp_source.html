11// ===========================================================================
12// Typed RELAY lifecycle-state enums (Tier-3 idiom pass).
13//
14// The Python reference + this port model call / dial / message lifecycle as
15// bare-string constants (`CALL_STATE_*`, `dial_state`, `MESSAGE_STATE_*` in
16// `relay/constants.py` / `relay/constants.hpp`) with stringly predicates
17// (`is_answered`, `is_ended`, `is_delivered`, …). These are KNOWABLE closed
18// sets, so a statically-typed port should expose them as `enum class`
19// ALONGSIDE the existing string accessors (parity), per the floor-not-ceiling
20// rule.
21//
22// ★ THREE SEPARATE VOCABULARIES that must NEVER be conflated — each mirrors a
23// distinct field the server emits:
24// * CallState — `call_state` on calling.call.state events
25// {created, ringing, answered, ending, ended}
26// * DialState — `dial_state` on the calling.call.dial event
27// {dialing, answered, failed} (RELAY_IMPLEMENTATION_GUIDE.md
28// line 193: "dialing (progress), answered (success),
29// failed (all legs failed)"). NOTE the dial OUTCOME set is
30// distinct from CallState — `failed` is dial-only and there
31// is no dial `created`/`ringing`/`ending`.
32// * MessageState — `message_state` on messaging.state events
33// {queued, initiated, sent, delivered, undelivered, failed,
34// received} (relay/constants.py MESSAGE_STATE_*).
35//
36// These mirror SERVER-emitted values that can GROW, so `*_from_string` returns
37// `std::optional<E>` (empty on an unknown value — never throws); the typed
38// accessors below stay usable even when the server introduces a new state.
39// `*_value()` / `to_string` are the single normalization point: the enum and
40// the bare string round-trip to the IDENTICAL wire string.
41//
42// `is_terminal()` follows the reference's own definition of "terminal"
43// (ended/failed — the states after which no further transitions occur):
44// * CallState::Ended
45// * DialState::Answered | DialState::Failed (the dial RPC resolves on either)
46// * MessageState::{Delivered, Undelivered, Failed} (== MESSAGE_TERMINAL_STATES)
47//
48// `[[nodiscard]]` on the value-returning free functions (consistent with the
49// Tier-2 pass): a stray `call_state_value(x);` that drops the result is a bug.
50// ===========================================================================
51
52// ---------------------------------------------------------------------------
53// CallState — calling.call.state `call_state`
54// ---------------------------------------------------------------------------
55
59enum class CallState {
60 Created,
61 Ringing,
62 Answered,
63 Ending,
64 Ended,
65};
66
67[[nodiscard]] inline std::string call_state_value(CallState v) {
68 switch (v) {
69 case CallState::Created: return "created";
70 case CallState::Ringing: return "ringing";
71 case CallState::Answered: return "answered";
72 case CallState::Ending: return "ending";
73 case CallState::Ended: return "ended";
74 }
75 return ""; // unreachable for a valid enumerator
76}
77
79[[nodiscard]] inline std::string to_string(CallState v) { return call_state_value(v); }
80
83[[nodiscard]] inline std::optional<CallState> call_state_from_string(const std::string& s) {
84 if (s == "created") return CallState::Created;
85 if (s == "ringing") return CallState::Ringing;
86 if (s == "answered") return CallState::Answered;
87 if (s == "ending") return CallState::Ending;
88 if (s == "ended") return CallState::Ended;
89 return std::nullopt;
90}
91
93[[nodiscard]] inline bool is_terminal(CallState v) { return v == CallState::Ended; }
