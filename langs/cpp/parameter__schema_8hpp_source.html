16// ===========================================================================
17// ParameterSchema — a typed, fluent builder for SWAIG tool parameters.
18//
19// `define_tool(name, desc, parameters, handler)` / `swaig::ToolDefinition`
20// take `parameters` as a raw `json` blob: a JSON-Schema `object` with a
21// `properties` map (one entry per argument) and an optional `required` list,
22// hand-written as nested `json::object(...)`:
23//
24// json::object({{"type", "object"}, {"properties", json::object({
25// {"query", json::object({{"type","string"},{"description","..."}})}
26// })}, {"required", json::array({"query"})}})
27//
28// That hand-written form is error-prone (typo'd keys, mismatched braces, a
29// `"type"` value that isn't a real JSON-Schema type, an `enum` written as the
30// wrong JSON kind). `ParameterSchema` is a TYPED CONVENIENCE OVER THE SAME
31// WIRE OUTPUT — it builds the IDENTICAL `json` value, byte-for-byte, via a
32// fluent API:
33//
34// swaig::ParameterSchema{}
35// .string("service", "The service to book")
36// .string("date", "YYYY-MM-DD")
37// .enum_of("fmt", swaig::record_format_values(), "Recording format")
38// .required({"service", "date"})
39// .to_json();
40//
41// `to_json()` returns exactly what the hand-written `json::object(...)` above
42// would — same keys, same values, same `required` array. (nlohmann::json
43// serialises object keys in sorted order, so the `.dump()` is canonical and
44// independent of insertion order; see the byte-identical proof tests.)
45//
46// This is ADDITIVE: the untyped `json` path keeps working unchanged. The
47// builder is a PORT_ADDITION (Python's `define_tool` takes a raw dict; there
48// is no typed builder in the reference). Idiom: "let the language shine" —
49// type the knowable JSON-Schema property shapes that Python leaves as a bare
50// dict (porting-sdk/IDIOM_PASS_JOURNAL.md §4 "Tier 2 flagship").
51//
52// `[[nodiscard]]` policy (consistent with the earlier Tier-2 pass + DataMap):
53// the value-returning terminals (`to_json()`, const accessors) are
54// `[[nodiscard]]` — discarding the built schema is a bug. The fluent setters
55// that return `ParameterSchema&` for chaining are intentionally NOT nodiscard
56// (chaining + a final `to_json()` is the whole point).
57// ===========================================================================

93 // -- Scalar property kinds ----------------------------------------------
94 // Each adds one entry to `properties`: a JSON-Schema object whose `type`
95 // is the JSON-Schema scalar type, plus an optional `description`.
96
98 ParameterSchema& string(const std::string& name,
99 const std::string& description = "") {
100 return add_scalar(name, "string", description);
101 }
102
104 ParameterSchema& number(const std::string& name,
105 const std::string& description = "") {
106 return add_scalar(name, "number", description);
107 }
108
110 ParameterSchema& integer(const std::string& name,
111 const std::string& description = "") {
112 return add_scalar(name, "integer", description);
113 }
114
116 ParameterSchema& boolean(const std::string& name,
117 const std::string& description = "") {
118 return add_scalar(name, "boolean", description);
119 }
120
121 // -- Enum (closed set) ---------------------------------------------------
122
129 ParameterSchema& enum_of(const std::string& name,
130 const std::vector<std::string>& values,
131 const std::string& description = "") {
132 json prop = json::object();
133 prop["type"] = "string";
134 if (!description.empty()) prop["description"] = description;
135 json arr = json::array();
136 for (const auto& v : values) arr.push_back(v);
137 prop["enum"] = arr;
138 properties_[name] = prop;
139 return *this;
140 }
141
142 // -- Array ---------------------------------------------------------------
143
147 ParameterSchema& array_of(const std::string& name,
148 const std::string& item_type,
149 const std::string& description = "") {
150 json prop = json::object();
151 prop["type"] = "array";
152 if (!description.empty()) prop["description"] = description;
153 prop["items"] = json::object({{"type", item_type}});
154 properties_[name] = prop;
155 return *this;
156 }
157
161 ParameterSchema& array_of(const std::string& name,
162 const ParameterSchema& item_schema,
163 const std::string& description = "") {
164 json prop = json::object();
165 prop["type"] = "array";
166 if (!description.empty()) prop["description"] = description;
167 prop["items"] = item_schema.to_json();
168 properties_[name] = prop;
169 return *this;
170 }
171
172 // -- Object (nested) -----------------------------------------------------
173
179 ParameterSchema& object_of(const std::string& name,
180 const ParameterSchema& nested,
181 const std::string& description = "") {
182 json prop = nested.to_json(); // already {type:object, properties:{...}, [required]}
183 if (!description.empty()) prop["description"] = description;
184 properties_[name] = prop;
185 return *this;
186 }
187
188 // -- Escape hatch --------------------------------------------------------
189
195 ParameterSchema& property(const std::string& name, const json& schema) {
196 properties_[name] = schema;
197 return *this;
198 }
199
200 // -- Required ------------------------------------------------------------
201
205 ParameterSchema& required(const std::vector<std::string>& names) {
206 required_ = names;
207 return *this;
208 }
209
211 ParameterSchema& require(const std::string& name) {
212 required_.push_back(name);
213 return *this;
214 }
215
216 // -- Build ---------------------------------------------------------------
217
225 [[nodiscard]] json to_json() const {
226 json out = json::object();
227 out["type"] = "object";
228 out["properties"] = properties_;
229 if (!required_.empty()) {
230 json req = json::array();
231 for (const auto& r : required_) req.push_back(r);
232 out["required"] = req;
233 }
234 return out;
235 }
236
241 operator json() const { return to_json(); } // NOLINT(google-explicit-constructor)
242
243 // -- Accessors -----------------------------------------------------------
244
246 [[nodiscard]] bool empty() const { return properties_.empty(); }
247
249 [[nodiscard]] std::size_t size() const { return properties_.size(); }
250
251private:
252 ParameterSchema& add_scalar(const std::string& name,
253 const char* json_type,
254 const std::string& description) {
255 json prop = json::object();
256 prop["type"] = json_type;
257 if (!description.empty()) prop["description"] = description;
258 properties_[name] = prop;
259 return *this;
260 }
261
262 json properties_ = json::object();
263 std::vector<std::string> required_;
264};
