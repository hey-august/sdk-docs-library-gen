14inline LogLevel get_log_level() {
15 std::string level = "";
16 const char* env = std::getenv("SIGNALWIRE_LOG_LEVEL");
17 if (env) level = env;
18 const char* mode = std::getenv("SIGNALWIRE_LOG_MODE");
19 if (mode && std::string(mode) == "off") return LogLevel::OFF;
20 if (level == "debug") return LogLevel::DEBUG;
21 if (level == "warn") return LogLevel::WARN;
22 if (level == "error") return LogLevel::ERROR;
23 return LogLevel::INFO;
24}
