13enum class LogLevel { DEBUG, INFO, WARN, ERROR, OFF };

15[[nodiscard]] inline LogLevel get_log_level() {
16 std::string level = "";
17 const char* env = std::getenv("SIGNALWIRE_LOG_LEVEL");
18 if (env) level = env;
19 const char* mode = std::getenv("SIGNALWIRE_LOG_MODE");
20 if (mode && std::string(mode) == "off") return LogLevel::OFF;
21 if (level == "debug") return LogLevel::DEBUG;
22 if (level == "warn") return LogLevel::WARN;
23 if (level == "error") return LogLevel::ERROR;
24 return LogLevel::INFO;
25}

27class Logger {
28public:

31 // `msg` is consumed read-only (streamed to cerr, never stored), so it
32 // takes a std::string_view — callers can pass a std::string, a literal,
33 // or a substring view with no allocation. (The constructor's `name`
34 // stays std::string: it is retained in name_.)
35 void debug(std::string_view msg) const {
36 if (get_log_level() <= LogLevel::DEBUG)
37 std::cerr << "[DEBUG][" << name_ << "] " << msg << std::endl;
38 }
39 void info(std::string_view msg) const {

42 }
