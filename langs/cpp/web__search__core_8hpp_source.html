5// Shared core for the web_search skill's latency-control + result formatting.
6//
7// Ports Python signalwire/skills/web_search/skill.py (51101da + 295745b):
8// per_page_timeout — per-page fetch timeout (the HTTP client's per-request
9// timeout). Default 2.0s.
10// overall_deadline — wall-clock budget for the whole tool call, enforced
11// with std::chrono::steady_clock. In-flight scrapes are
12// abandoned once it fires. Default 10.0s. THIS IS THE
13// CONTRACT — a slow site must never blow past the kernel
14// webhook timeout (~55s).
15// parallel_scrape — scrape all candidate result links concurrently via
16// std::async, harvesting whatever finishes before the
17// deadline (best-effort, NOT contracted). Default true.
18// snippets_only — skip scraping entirely and format the CSE snippets
19// directly. Sub-second. Default false.
20// + snippet fallback — when no scraped result survives (every page failed,
21// was empty, or the deadline fired) we format the CSE
22// snippets we already have rather than returning an empty
23// "no results" message, so the model always gets useful
24// context back.
25//
26// This header is included by BOTH web_search implementations that exist in the
27// tree — the registered `WebSearchSkillR` in skill_registry.cpp and the
28// `WebSearchSkill` in builtin/web_search.cpp — so the latency contract lives in
29// exactly one place and the two skills can never drift. (The builtin's
30// REGISTER_SKILL is dead-stripped from the static archive; WebSearchSkillR is
31// what actually runs. Both delegate here.)
32

55struct Candidate {
56 std::string title;
57 std::string link;
58 std::string snippet;
59 std::string content; // populated only after a successful scrape
60 bool scraped = false; // true once `content` holds a usable page body
61};

64struct LatencyParams {
65 double per_page_timeout = 2.0; // seconds
66 double overall_deadline = 10.0; // seconds
67 bool parallel_scrape = true;
68 bool snippets_only = false;

73[[nodiscard]] inline std::vector<Candidate> parse_cse_items(const json& parsed) {
74 std::vector<Candidate> out;
75 if (parsed.contains("items") && parsed["items"].is_array()) {
76 for (const auto& it : parsed["items"]) {
77 Candidate c;
78 c.title = it.value("title", "");
79 c.link = it.value("link", "");
80 c.snippet = it.value("snippet", "");
81 out.push_back(std::move(c));
82 }
83 }
84 return out;
85}

91[[nodiscard]] inline std::string format_snippet_results(const std::string& query,
92 const std::vector<Candidate>& cands,
93 int num_results) {
94 std::size_t top = static_cast<std::size_t>(num_results < 1 ? 1 : num_results);
95 top = std::min(top, cands.size());
96 std::ostringstream out;
97 out << "Snippet-only results for '" << query
98 << "' (page content not scraped):\n\n";
99 for (std::size_t i = 0; i < top; ++i) {
100 const auto& c = cands[i];
101 std::string snip = c.snippet;
102 // Trim leading/trailing whitespace (mirrors Python's .strip()).
103 auto b = snip.find_first_not_of(" \t\r\n");
104 auto e = snip.find_last_not_of(" \t\r\n");
105 if (b == std::string::npos) snip.clear();
106 else snip = snip.substr(b, e - b + 1);
107 out << "=== RESULT " << (i + 1) << " ===\n"
108 << "Title: " << c.title << "\n"
109 << "URL: " << c.link << "\n"
110 << "Snippet: " << snip << "\n\n";
111 }
112 std::string s = out.str();
113 // Drop the trailing blank line so wrapping is tidy.
114 while (!s.empty() && (s.back() == '\n')) s.pop_back();
115 return s;
116}

122[[nodiscard]] inline std::string format_scraped_results(const std::string& query,
123 const std::vector<Candidate>& scraped,
124 std::size_t total_candidates,
125 std::size_t per_result_limit) {
126 std::ostringstream out;
127 out << "Web search results for '" << query << "':\n";
128 out << "Showing " << scraped.size() << " of " << total_candidates
129 << " results (page content scraped):\n\n";
130 int idx = 1;
131 for (const auto& c : scraped) {
132 out << "=== RESULT " << idx++ << " ===\n"
133 << "Title: " << c.title << "\n"
134 << "URL: " << c.link << "\n"
135 << "Snippet: " << c.snippet << "\n"
136 << "Content:\n";
137 std::string content = c.content;
138 if (content.size() > per_result_limit) {
139 content = content.substr(0, per_result_limit) + "...";
140 }
141 out << content << "\n"
142 << "==================================================\n\n";
143 }
144 std::string s = out.str();
145 while (!s.empty() && (s.back() == '\n')) s.pop_back();
146 return s;
147}

153[[nodiscard]] inline Candidate scrape_one(const Candidate& in, double per_page_timeout) {
154 Candidate c = in;
155 if (c.link.empty()) return c;
156 long timeout_ms = static_cast<long>(per_page_timeout * 1000.0);
157 // Belt-and-suspenders: http_get_ms already converts transport/scheme
158 // failures into status 0, but guard here too so a scrape can never throw
159 // out of the std::async task (which would terminate the process).
160 try {
161 auto resp = http_get_ms(c.link, {}, timeout_ms);
162 if (resp.status >= 200 && resp.status < 300 && !resp.body.empty()) {
163 c.content = resp.body;
164 c.scraped = true;
165 }
166 } catch (...) {
167 // leave c unscraped
168 }
169 return c;
170}

186[[nodiscard]] inline std::vector<Candidate> scrape_candidates(const std::vector<Candidate>& cands,
187 const LatencyParams& lp,
188 steady_clock::time_point deadline_at) {
189 std::vector<Candidate> scraped;
190
191 if (!lp.parallel_scrape) {
192 for (const auto& c : cands) {
193 if (steady_clock::now() >= deadline_at) break; // out of budget
194 Candidate r = scrape_one(c, lp.per_page_timeout);
195 if (r.scraped) scraped.push_back(std::move(r));
196 }
197 return scraped;
198 }
199
200 // Parallel: launch all scrapes, then harvest whatever finishes before the
201 // deadline.
202 std::vector<std::future<Candidate>> futures;
203 futures.reserve(cands.size());
204 double ppt = lp.per_page_timeout;
205 for (const auto& c : cands) {
206 futures.push_back(std::async(std::launch::async,
207 [c, ppt]() { return scrape_one(c, ppt); }));
208 }
209
210 // Anything still running at the deadline is moved here and reaped by a
211 // detached thread so the abandoned future's blocking destructor runs OFF
212 // the caller's path. The fetch is per_page_timeout-bounded, so the reaper
213 // terminates promptly on its own.
214 auto leftovers = std::make_shared<std::vector<std::future<Candidate>>>();
215
216 for (auto& f : futures) {
217 auto now = steady_clock::now();
218 if (now >= deadline_at) {
219 // Deadline already blown — abandon this and everything after it.
220 if (f.valid()) leftovers->push_back(std::move(f));
221 continue;
222 }
223 if (f.wait_until(deadline_at) == std::future_status::ready) {
224 Candidate r = f.get();
225 if (r.scraped) scraped.push_back(std::move(r));
226 } else {
227 // Timed out at the deadline — abandon.
228 if (f.valid()) leftovers->push_back(std::move(f));
229 }
230 }
231
232 if (!leftovers->empty()) {
233 std::thread([leftovers]() {
234 for (auto& f : *leftovers) {
235 if (f.valid()) f.wait(); // bounded by per_page_timeout
236 }
237 }).detach();
238 }
239
240 return scraped;
241}

254[[nodiscard]] inline std::string run(const std::string& query,
255 std::vector<Candidate> candidates,
256 const LatencyParams& lp,
257 int num_results,
258 std::size_t max_content_length,
259 const std::string& empty_no_items_message) {
260 if (candidates.empty()) {
261 return empty_no_items_message;
262 }
263
264 // snippets_only fast path: skip scraping entirely.
265 if (lp.snippets_only) {
266 return format_snippet_results(query, candidates, num_results);
267 }
268
269 // overall_deadline budget starts now (after the non-cancelable CSE fetch,
270 // which the caller already did). steady_clock is monotonic — immune to
271 // wall-clock jumps.
272 auto deadline_at = steady_clock::now() +
273 std::chrono::milliseconds(static_cast<long>(lp.overall_deadline * 1000.0));
274
275 std::vector<Candidate> scraped = scrape_candidates(candidates, lp, deadline_at);
276
277 if (scraped.empty()) {
278 // Time ran out or every page failed/was below threshold. Fall back to
279 // CSE snippets so we return SOMETHING useful before the kernel webhook
280 // timeout fires (Python parity).
281 return format_snippet_results(query, candidates, num_results);
282 }
283
284 // Keep at most num_results scraped survivors.
285 std::size_t keep = static_cast<std::size_t>(num_results < 1 ? 1 : num_results);
286 if (scraped.size() > keep) scraped.resize(keep);
287
288 // Per-result content budget (mirrors the spirit of Python's calculation).
289 std::size_t overhead = scraped.size() * 400;
290 std::size_t available = max_content_length > overhead
291 ? max_content_length - overhead
292 : max_content_length;
293 std::size_t per_result_limit = scraped.empty() ? 2000 : available / scraped.size();
294 if (per_result_limit < 2000) per_result_limit = 2000;
295
296 return format_scraped_results(query, scraped, candidates.size(), per_result_limit);
297}
