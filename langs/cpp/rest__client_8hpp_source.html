277 // ----------------------------------------------------------------
278 // Command-dispatch entry point (matches Python CallingNamespace).
279 //
280 // All 32 commands are POST /api/calling/calls with a body shaped
281 // {"command": "<name>", "id": "<call_id>"?, "params": {...}}.
282 // ``id`` is omitted for calls that target no specific call (e.g.
283 // ``dial`` and ``update``).
284 // ----------------------------------------------------------------
285 json execute(const std::string& command,
286 const json& params,
287 const std::optional<std::string>& call_id = std::nullopt) const {
288 json body = {{"command", command}, {"params", params}};
289 if (call_id) body["id"] = *call_id;
290 return client.post("/api/calling/calls", body);
291 }
