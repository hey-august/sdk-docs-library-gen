13// ===========================================================================
14// join_conference closed-set enums
15//
16// Mirrors the Gender / SkillName affordance pattern: each closed set the
17// Python reference documents as a bare string gets a typed `enum class` so a
18// typo fails at the call site instead of on the server. The string form still
19// works (Python uses bare `str`/`Optional[str]`), so engine-/CXML-specific
20// values that aren't one of the canonical members remain expressible. The
21// `*_value()` helpers are the single normalization point — enum and string
22// emit the identical wire string.
23// ===========================================================================
24
26enum class ConferenceBeep { True, False, OnEnter, OnExit };
28enum class ConferenceRecord { DoNotRecord, RecordFromStart };
30enum class ConferenceTrim { TrimSilence, DoNotTrim };
32enum class CallbackMethod { Get, Post };

34// The closed-set value mappers below are pure functions whose only effect is
35// the returned wire string — [[nodiscard]] so a stray `conference_beep_value(x);`
36// (forgetting to use the result) is flagged.

45}

68// ===========================================================================
69// record_call / tap media closed-set enums
70//
71// Same SkillName / Gender / join_conference affordance: the Python reference
72// validates each of these against a fixed set in `core/function_result.py`
73// (`raise ValueError` on a miss), so a typo only fails at runtime on the
74// server. The typed `enum class` makes the typo fail at the call site with
75// editor autocompletion, while the bare-`std::string` overloads stay canonical
76// (parity with Python's bare `str`). The `*_value()` helpers are the single
77// normalization point — the enum overload routes its wire string into the
78// EXACT string method, so the emitted SWML is byte-identical.
79//
80// ★ Three direction vocabularies + two codec vocabularies that must NEVER be
81// unified (they are bug generators): record_call's direction is
82// {speak,listen,both}, tap's direction is {speak,hear,both} (`hear`, not
83// `listen`), and tap's codec is the 2-value SWAIG set {PCMU,PCMA} — NOT the
84// wider RELAY connect/stream codec superset. Each set gets its OWN enum,
85// faithfully mirroring the reference's separate validation lists.
86// ===========================================================================
87
90enum class RecordFormat { Wav, Mp3, Mp4 };
91
95enum class RecordDirection { Speak, Listen, Both };
96
100enum class TapDirection { Speak, Hear, Both };
101
105enum class Codec { Pcmu, Pcma };

154template <typename E, std::string (*Map)(E)>
155struct EnumOrString {
156 std::string value;
157 EnumOrString(E e) : value(Map(e)) {} // NOLINT(google-explicit-constructor)
158 EnumOrString(const std::string& s) : value(s) {} // NOLINT
159 EnumOrString(const char* s) : value(s) {} // NOLINT
160 [[nodiscard]] const std::string& str() const { return value; }
161};
