5// Shared TLS cert-resolver + httplib server factory for the SDK's three
6// in-process HTTP servers (AgentServer, swml::Service, agent::AgentBase).
7//
8// When TLS is configured the factory returns an httplib::SSLServer (which
9// IS-A httplib::Server, so it upcasts cleanly into the existing
10// std::unique_ptr<httplib::Server> every server site already holds, and
11// setup_routes() is unchanged). Otherwise it returns a plain Server.
12//
13// Configuration mirrors the Python reference (security_config.py):
14// SWML_SSL_ENABLED — "true"/"1"/"yes" turns TLS on
15// SWML_SSL_CERT_PATH — server certificate (PEM)
16// SWML_SSL_KEY_PATH — server private key (PEM)
17// An explicit per-instance setter (TlsServerConfig passed directly) overrides
18// the environment so callers can enable TLS programmatically.

30struct TlsServerConfig {
31 bool enabled = false;
32 std::string cert_path;
33 std::string key_path;
34
36 bool usable() const {
37 return enabled && !cert_path.empty() && !key_path.empty();
38 }

44inline TlsServerConfig resolve_tls_config_from_env() {
45 TlsServerConfig cfg;
46 std::string enabled = get_env("SWML_SSL_ENABLED", "");
47 for (auto& c : enabled) c = static_cast<char>(::tolower(c));
48 cfg.enabled = (enabled == "true" || enabled == "1" || enabled == "yes" || enabled == "on");
49 cfg.cert_path = get_env("SWML_SSL_CERT_PATH", "");
50 cfg.key_path = get_env("SWML_SSL_KEY_PATH", "");
51 return cfg;
52}

60inline std::unique_ptr<httplib::Server> make_http_server(const TlsServerConfig& cfg) {
61 if (cfg.usable()) {
62 return std::make_unique<httplib::SSLServer>(cfg.cert_path.c_str(),
63 cfg.key_path.c_str());
64 }
65 return std::make_unique<httplib::Server>();
66}
