34// Generate a simple UUID-like string
35inline std::string generate_uuid() {
36 static std::random_device rd;
37 static std::mt19937 gen(rd());
38 static std::uniform_int_distribution<uint32_t> dis;
39 auto r = [&]() { return dis(gen); };
40 char buf[40];
41 std::snprintf(buf, sizeof(buf),
42 "%08x-%04x-%04x-%04x-%04x%08x",
43 r(), r() & 0xffff, (r() & 0x0fff) | 0x4000,
44 (r() & 0x3fff) | 0x8000, r() & 0xffff, r());
45 return std::string(buf);
46}

54// URL encode a string
55inline std::string url_encode(const std::string& value) {
56 std::ostringstream escaped;
57 escaped.fill('0');
58 escaped << std::hex;
59 for (char c : value) {
60 if (isalnum(static_cast<unsigned char>(c)) || c == '-' || c == '_' || c == '.' || c == '~') {
61 escaped << c;
62 } else {
63 escaped << '%' << std::setw(2) << std::uppercase << int(static_cast<unsigned char>(c));
64 }
65 }
66 return escaped.str();
67}

69// Generate a random password
70inline std::string generate_random_password(int length = 32) {
71 static const char charset[] =
72 "abcdefghijklmnopqrstuvwxyz"
73 "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
74 "0123456789";
75 std::random_device rd;
76 std::mt19937 gen(rd());
77 std::uniform_int_distribution<> dis(0, sizeof(charset) - 2);
78 std::string result;
79 result.reserve(length);
80 for (int i = 0; i < length; ++i) {
81 result += charset[dis(gen)];
82 }
83 return result;
84}

86// Timing-safe string comparison
87inline bool timing_safe_compare(const std::string& a, const std::string& b) {
88 if (a.size() != b.size()) return false;
89 volatile unsigned char result = 0;
90 for (size_t i = 0; i < a.size(); ++i) {
91 result |= static_cast<unsigned char>(a[i]) ^ static_cast<unsigned char>(b[i]);
92 }
93 return result == 0;
94}

96// Base64 encode
97inline std::string base64_encode(const std::string& input) {
98 static const char table[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
99 std::string out;
100 int val = 0, valb = -6;
101 for (unsigned char c : input) {
102 val = (val << 8) + c;
103 valb += 8;
104 while (valb >= 0) {
105 out.push_back(table[(val >> valb) & 0x3F]);
106 valb -= 6;
107 }
108 }
109 if (valb > -6) out.push_back(table[((val << 8) >> (valb + 8)) & 0x3F]);
110 while (out.size() % 4) out.push_back('=');
111 return out;
112}
