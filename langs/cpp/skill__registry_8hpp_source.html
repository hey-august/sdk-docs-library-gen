65 void add_skill_directory(const std::string& path) {
66 std::lock_guard<std::mutex> lock(mutex_);
67 struct stat st;
68 if (::stat(path.c_str(), &st) != 0) {
69 throw std::invalid_argument("Skill directory does not exist: " + path);
70 }
71 if (!S_ISDIR(st.st_mode)) {
72 throw std::invalid_argument("Path is not a directory: " + path);
73 }
74 for (const auto& existing : external_paths_) {
75 if (existing == path) return; // already present, dedup
76 }
77 external_paths_.push_back(path);
78 }
