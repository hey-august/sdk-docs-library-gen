44struct LanguageConfig {
45 std::string name;
46 std::string code;
47 std::string voice;
48 std::string engine;
49 std::string fillers;
55 json params;
56
57 [[nodiscard]] json to_json() const {
58 json j;
59 j["name"] = name;
60 j["code"] = code;
61 j["voice"] = voice;
62 if (!engine.empty()) j["engine"] = engine;
63 if (!fillers.empty()) j["fillers"] = fillers;
64 // Only emit the params key when non-empty so we don't pollute
65 // SWML with empty objects (parity with Python's
66 // ``if params:`` check).
67 if (params.is_object() && !params.empty()) {
68 j["params"] = params;
69 }
70 return j;
71 }
72};

462 //
463 // When ``signing_key`` is set, the agent server auto-mounts the
464 // webhook validator on POST `/`, `/swaig`, and `/post_prompt`. Unsigned
465 // or wrongly-signed requests get a 403 and never reach the handler.
466 //
467 // Resolution order at runtime:
468 // 1. ``set_signing_key(...)`` (explicit)
