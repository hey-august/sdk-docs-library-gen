43struct LanguageConfig {
44 std::string name;
45 std::string code;
46 std::string voice;
47 std::string engine;
48 std::string fillers;
49
50 json to_json() const {
51 json j;
52 j["name"] = name;
53 j["code"] = code;
54 j["voice"] = voice;

419 // ========================================================================
420 // Webhook Signature Validation (porting-sdk/webhooks.md)
421 //
422 // When ``signing_key`` is set, the agent server auto-mounts the
423 // webhook validator on POST `/`, `/swaig`, and `/post_prompt`. Unsigned
424 // or wrongly-signed requests get a 403 and never reach the handler.
425 //
426 // Resolution order at runtime:
427 // 1. ``set_signing_key(...)`` (explicit)
428 // 2. ``SIGNALWIRE_SIGNING_KEY`` env var (fallback)
429 // When neither is set, AgentBase logs a startup warning matching the
430 // Python reference and accepts unsigned POSTs.
431 // ========================================================================
